## 矩阵与矩阵运算


### 一句话

给定一个 $k$ 阶常系数线性递推 $f_n=\sum_{i=1}^{k}c_if_{n-i}$ 与前 $k$ 项（$k\le 100$），要你求出第 $n$ 项 $f_n$（$n\le 10^{18}$），$k$ 很小而 $n$ 极大，同一组系数要反复用来推后面的项。**时间 $O(k^3\log n)$，空间 $O(k^2)$。**


### 原理


### 实现

```c++
const int SZ = 105;
const ll MOD = 998244353;

struct Mat {
    int n, m;
    ll a[SZ][SZ];
    Mat(int n_ = 0, int m_ = 0) : n(n_), m(m_) { memset(a, 0, sizeof a); }
    static Mat id(int n) {
        Mat r(n, n);
        for (int i = 0; i < n; i++) r.a[i][i] = 1;
        return r;
    }
    Mat operator*(const Mat& o) const {
        Mat r(n, o.m);
        for (int i = 0; i < n; i++)
            for (int k = 0; k < m; k++) {
                if (!a[i][k]) continue;
                for (int j = 0; j < o.m; j++)
                    r.a[i][j] = (r.a[i][j] + a[i][k] * o.a[k][j]) % MOD;
            }
        return r;
    }
};
Mat mat_pow(Mat b, ll e) {
    Mat r = Mat::id(b.n);
    for (; e; e >>= 1, b = b * b)
        if (e & 1) r = r * b;
    return r;
}
```


### 应用


#### 斐波那契

$\begin{pmatrix}1&1\\1&0\end{pmatrix}^n$ 的右上角就是 $F_n$；把状态向量取成 $(F_n,F_{n-1})$，转移矩阵第一行放递推系数、下面放一条斜对角即可。

```c++
ll fib_mat(ll n) {
    Mat b(2, 2);
    b.a[0][0] = 1, b.a[0][1] = 1, b.a[1][0] = 1, b.a[1][1] = 0;
    return mat_pow(b, n).a[0][1];
}
// F(0)=0, F(1)=1：fib_mat(0)=0, fib_mat(1)=1, fib_mat(10)=55
```


#### 带常数项的递推

把常量 $1$ 也放进状态向量（矩阵多一行一列，最后一行是 $(0,\dots,0,1)$），就能处理 $f_n=\sum c_if_{n-i}+c_0$。

```c++
ll affine_rec(ll a, ll b, ll n0) {           // f(n) = a*f(n-1) + b
    Mat t(2, 2);
    t.a[0][0] = a, t.a[0][1] = b;
    t.a[1][1] = 1;
    return mat_pow(t, n0).a[0][1];           // 从 f(0)=0 出发
}
// 例：f(n) = 2f(n-1) + 1，f(0)=0 => f(n) = 2^n - 1
```


#### 图上路径计数（走 $k$ 步）

邻接矩阵的 $k$ 次幂，$A^k_{ij}$ 就是「从 $i$ 走恰好 $k$ 步到 $j$ 的方案数」。

```c++
ll count_walks(int n_, vector<vector<int>>& adj, ll k, int s, int t) {
    Mat A(n_, n_);
    for (int u = 0; u < n_; u++)
        for (int v : adj[u]) A.a[u][v] = (A.a[u][v] + 1) % MOD;   // 重边累加
    return mat_pow(A, k).a[s][t];            // 恰好 k 步的方案数
}
// 复杂度 O(n^3 log k)；只问单点对时把初始向量乘进去可以省成 O(n^2 log k)
```


#### 广义矩阵（max-plus）

把「乘法换成加法、加法换成取 max」，矩阵快速幂就能加速最长路型的 DP（Floyd 的广义版本）。

```c++
struct MaxMat {
    int n; ll a[SZ][SZ];
    MaxMat(int n_ = 0) : n(n_) { for (int i = 0; i < SZ; i++) for (int j = 0; j < SZ; j++) a[i][j] = -(ll)4e18; }
    MaxMat operator*(const MaxMat& o) const {
        MaxMat r(n);
        for (int i = 0; i < n; i++)
            for (int k = 0; k < n; k++) {
                if (a[i][k] < -(ll)1e18) continue;
                for (int j = 0; j < n; j++)
                    if (o.a[k][j] > -(ll)1e18)
                        r.a[i][j] = max(r.a[i][j], a[i][k] + o.a[k][j]);
            }
        return r;
    }
};
// 用它就能求「恰好 k 条边的最长路」；把 max 换成 min 就是「恰好 k 条边的最短路」
```


## 高斯消元


### 一句话

给定一个 $n$ 元线性方程组 $Ax=b$ 和它的 $n$ 阶系数方阵（$n\le 500$），要你判断这个方程组是有唯一解、无解还是有无穷多解并求出一组解，同时算出方阵的**行列式**与**逆矩阵**，围绕同一个矩阵的几个问题要一起回答。**时间 $O(n^3)$，空间 $O(n^2)$。**


### 原理


### 实现

```c++
const double EPS = 1e-9;
double a[SZ][SZ];                            // 增广矩阵，第 n 列是 b

// ---------- 高斯消元：返回 -1 无穷多解，0 无解，1 唯一解（解写在 a[i][n]） ----------
int gauss(int n) {
    int r = 0;
    for (int c = 0; c < n && r < n; c++) {
        int p = r;
        for (int i = r + 1; i < n; i++)
            if (fabs(a[i][c]) > fabs(a[p][c])) p = i;
        if (fabs(a[p][c]) < EPS) continue;
        swap(a[r], a[p]);
        double inv = 1.0 / a[r][c];
        for (int j = c; j <= n; j++) a[r][j] *= inv;   // 归一化，让主元变成 1
        for (int i = 0; i < n; i++) {
            if (i == r || fabs(a[i][c]) < EPS) continue;
            double t = a[i][c];
            for (int j = c; j <= n; j++) a[i][j] -= t * a[r][j];
        }
        r++;
    }
    for (int i = r; i < n; i++)
        if (fabs(a[i][n]) > EPS) return 0;
    if (r < n) return -1;
    return 1;
}

// ---------- 行列式（模意义，用辗转相除避免求逆） ----------
ll det_mod(vector<vector<ll>> m, int n, ll mod) {
    ll res = 1;
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
            while (m[j][i]) {
                ll t = m[i][i] / m[j][i];
                for (int k = i; k < n; k++)
                    m[i][k] = (m[i][k] - (__int128)t * m[j][k]) % mod;
                swap(m[i], m[j]);
                res = (mod - res) % mod;
            }
        }
        if (m[i][i] == 0) return 0;
        res = res * ((m[i][i] % mod + mod) % mod) % mod;
    }
    return (res + mod) % mod;
}
```


### 应用


#### 矩阵求逆（模意义）

在右边拼一个单位矩阵一起消元；模意义下主元用扩展欧几里得求逆。

```c++
ll qpow(ll a, ll b, ll mod) {
    ll r = 1;
    for (a %= mod; b; b >>= 1, a = (__int128)a * a % mod)
        if (b & 1) r = (__int128)r * a % mod;
    return r;
}
bool mat_inv(vector<vector<ll>> m, vector<vector<ll>>& inv, int n, ll mod) {
    inv.assign(n, vector<ll>(n, 0));
    for (int i = 0; i < n; i++) inv[i][i] = 1;
    for (int c = 0; c < n; c++) {
        int p = -1;
        for (int i = c; i < n; i++) if (m[i][c]) { p = i; break; }
        if (p < 0) return false;             // 不可逆
        swap(m[c], m[p]), swap(inv[c], inv[p]);
        ll iv = qpow(m[c][c], mod - 2, mod);
        for (int j = 0; j < n; j++) m[c][j] = m[c][j] * iv % mod, inv[c][j] = inv[c][j] * iv % mod;
        for (int i = 0; i < n; i++) {
            if (i == c || !m[i][c]) continue;
            ll t = m[i][c];
            for (int j = 0; j < n; j++) {
                m[i][j] = (m[i][j] - t * m[c][j]) % mod;
                inv[i][j] = (inv[i][j] - t * inv[c][j]) % mod;
            }
        }
    }
    for (int i = 0; i < n; i++) for (int j = 0; j < n; j++) inv[i][j] = (inv[i][j] % mod + mod) % mod;
    return true;
}
```


#### 求矩阵的秩

消元后非零行的行数就是秩；也可以用来判断向量组是否线性相关。

```c++
int mat_rank(vector<vector<double>> m, int n, int cols) {
    int r = 0;
    for (int c = 0; c < cols && r < n; c++) {
        int p = r;
        for (int i = r + 1; i < n; i++) if (fabs(m[i][c]) > fabs(m[p][c])) p = i;
        if (fabs(m[p][c]) < EPS) continue;
        swap(m[r], m[p]);
        for (int i = r + 1; i < n; i++) {
            if (fabs(m[i][c]) < EPS) continue;
            double t = m[i][c] / m[r][c];
            for (int j = c; j < cols; j++) m[i][j] -= t * m[r][j];
        }
        r++;
    }
    return r;                                // 秩
}
```


#### 解异或方程组

把加减换成异或、系数取 0/1，用 bitset 压缩；秩为 $r$ 时自由元有 $n-r$ 个，解数就是 $2^{n-r}$。

```c++
// 异或方程组：每行是一个 bitset，最后一列是常数项
int gauss_xor(vector<bitset<1005>>& a, int n, int m) {   // m 个未知数
    int r = 0;
    for (int c = 0; c < m && r < n; c++) {
        int p = -1;
        for (int i = r; i < n; i++) if (a[i][c]) { p = i; break; }
        if (p < 0) continue;
        swap(a[r], a[p]);
        for (int i = 0; i < n; i++) if (i != r && a[i][c]) a[i] ^= a[r];
        r++;
    }
    for (int i = r; i < n; i++) if (a[i][m]) return -1;   // 0 = 1，无解
    return m - r;                            // 自由元个数；解的个数是 2^(m-r)
}
```


## 线性基（异或空间）


### 一句话

给定 $n$ 个非负整数（$n\le 10^5$，$a_i<2^{60}$），要你把这些数一个个插进一个结构里，再询问它们异或能表示出的最大值、第 $k$ 小值，以及某个数能否被表示，插入与询问交替出现。**时间 $O(60)$ 每次，空间 $O(60)$。**


### 原理


### 实现

```c++
struct LinearBasis {
    ll b[64];                                // b[i]：最高位为 i 的基向量
    LinearBasis() { memset(b, 0, sizeof b); }

    bool insert(ll x) {                      // 返回 x 是否「带来了新的基」
        for (int i = 62; i >= 0; i--)
            if (x >> i & 1) {
                if (!b[i]) { b[i] = x; return true; }
                x ^= b[i];
            }
        return false;
    }
    bool can(ll x) const {
        for (int i = 62; i >= 0; i--)
            if (x >> i & 1) {
                if (!b[i]) return false;
                x ^= b[i];
            }
        return true;
    }
    ll max_xor() const {
        ll r = 0;
        for (int i = 62; i >= 0; i--) r = max(r, r ^ b[i]);
        return r;
    }
    ll min_xor() const {
        for (int i = 0; i <= 62; i++) if (b[i]) return b[i];
        return 0;
    }
    // 第 k 小（k 从 1 开始，包含 0）：先把基化成最简形，再按 k-1 的二进制取
    // 注意：这里会修改 b，所以不能是 const 成员函数
    ll kth(ll k) {
        static ll c[64];
        int cnt = 0;
        for (int i = 0; i <= 62; i++)
            if (b[i]) {
                for (int j = i - 1; j >= 0; j--)
                    if (b[i] >> j & 1) b[i] ^= b[j];   // 消成最简形
                c[cnt++] = b[i];
            }
        --k;                                 // 空集给出 0，它就是第 1 小
        ll r = 0;
        for (int i = 0; i < cnt; i++)
            if (k >> i & 1) r ^= c[i];
        return r;
    }
};
```


### 应用


#### 最大异或和

从高位到低位贪心：能变大就异或上这一位的基。把两棵线性基「合并」时，把一棵的基逐个插进另一棵即可，这是线段树维护区间异或的基础。

```c++
LinearBasis merge_basis(const LinearBasis& x, const LinearBasis& y) {
    LinearBasis r = x;
    for (int i = 0; i <= 62; i++)
        if (y.b[i]) r.insert(y.b[i]);
    return r;
}
// 用它可以在线段树上维护「任意区间的线性基」，查询时把 O(log n) 个结点合并起来
// 每次合并 O(60 * 60)，整体 O(log n * 60^2)
```


#### 异或第 $k$ 小

把基化成最简形（每个基向量的最高位互不相同、且其他基在该位为 0），再按 $k-1$ 的二进制挑基向量。

```c++
// 用法示例：求出所有可表示异或值中第 k 小（含 0）
ll query_kth(LinearBasis lb, ll k) { return lb.kth(k); }
// 若要「非空子集」的第 k 小，把 k 换成 k+1；若集合本身线性相关（0 可表示），
// 要注意 k 的上界是 2^rank，越界说明不存在
// 例：基 {1,2,4}，k=1..8 依次得到 0,1,2,3,4,5,6,7
```


#### 区间最大异或和（前缀线性基）

对每个前缀维护一份线性基，并让每个基向量的「位置」尽量靠右；查询 $[l,r]$ 时只取位置 $\ge l$ 的基，$O(60)$ 一次。

```c++
int pos[64];                                 // 每个基向量对应的下标（尽量靠右）
ll bb[64];
void insert_at(ll x, int idx) {
    for (int i = 62; i >= 0; i--) {
        if (!(x >> i & 1)) continue;
        if (!bb[i]) { bb[i] = x, pos[i] = idx; return; }
        if (pos[i] < idx) { swap(bb[i], x), swap(pos[i], idx); }   // 让位置靠右的留下
        x ^= bb[i];
    }
}
ll query_max(int l) {                        // 只考虑位置 >= l 的基
    ll r = 0;
    for (int i = 62; i >= 0; i--)
        if (pos[i] >= l) r = max(r, r ^ bb[i]);
    return r;
}
// 每个前缀存一份 bb/pos 就能回答任意 [l, r]（可持久化版本）
```


#### 异或方程组的解数

$n$ 个方程 $m$ 个未知数，异或方程组消元后自由元有 $m-r$ 个，解的个数就是 $2^{m-r}$。

```c++
ll count_xor_solutions(vector<bitset<1005>> a, int n, int m) {
    int r = 0;
    for (int c = 0; c < m && r < n; c++) {
        int p = -1;
        for (int i = r; i < n; i++) if (a[i][c]) { p = i; break; }
        if (p < 0) continue;                 // 自由元
        swap(a[r], a[p]);
        for (int i = 0; i < n; i++) if (i != r && a[i][c]) a[i] ^= a[r];
        r++;
    }
    for (int i = r; i < n; i++) if (a[i][m]) return 0;   // 出现 0 = 1，无解
    return 1LL << (m - r);                   // 自由元个数是 m - r
}
```


## 矩阵树定理


### 一句话

给定一张 $n$ 个点、$m$ 条边的图（$n\le 500$，边可以带权），要你数出它一共有多少棵生成树（带权时数的是所有生成树的边权积之和），图一次给全。**时间 $O(n^3)$，空间 $O(n^2)$。**


### 原理


### 实现

（基尔霍夫矩阵 $L=D-A$；去掉任意一行一列后求行列式。）

```c++
ll det_mod(vector<vector<ll>> m, int n, ll mod) {   // 见「高斯消元与行列式」一节
    ll res = 1;
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
            while (m[j][i]) {
                ll t = m[i][i] / m[j][i];
                for (int k = i; k < n; k++)
                    m[i][k] = (m[i][k] - (__int128)t * m[j][k]) % mod;
                swap(m[i], m[j]);
                res = (mod - res) % mod;
            }
        }
        if (m[i][i] == 0) return 0;
        res = res * ((m[i][i] % mod + mod) % mod) % mod;
    }
    return (res + mod) % mod;
}
ll matrix_tree(int n, vector<array<ll, 3>>& edges, ll mod) {
    vector<vector<ll>> L(n, vector<ll>(n, 0));
    for (auto [u, v, w] : edges) {           // w 是边权（计数时取 1）
        L[u][u] = (L[u][u] + w) % mod, L[v][v] = (L[v][v] + w) % mod;
        L[u][v] = (L[u][v] - w % mod + mod) % mod;
        L[v][u] = (L[v][u] - w % mod + mod) % mod;
    }
    vector<vector<ll>> A(n - 1, vector<ll>(n - 1));
    for (int i = 0; i < n - 1; i++)
        for (int j = 0; j < n - 1; j++) A[i][j] = L[i][j];
    return det_mod(A, n - 1, mod);
}
```


### 应用


#### 生成树计数（无向图）

边权全取 $1$，答案就是生成树个数。

```c++
ll spanning_trees(int n, vector<pair<int, int>>& es, ll mod) {
    vector<array<ll, 3>> e;
    for (auto [u, v] : es) e.push_back({(ll)u, (ll)v, 1});
    return matrix_tree(n, e, mod);
}
// 完全图 K_n 上算出来应当等于 n^(n-2)（Cayley 公式）；不连通图算出来是 0
```


#### 带权生成树

边权 $w$ 时，矩阵树定理给出的是「所有生成树的 $\prod w_e$ 之和」，也就是把边权当作「并联的条数」。

```c++
ll weighted_trees(int n, vector<array<ll, 3>>& e, ll mod) { return matrix_tree(n, e, mod); }
// 平行边可以直接合并：两条 (u,v,2) 与 (u,v,3) 等价于一条 (u,v,5)
// 验证：K_2 上一条权 w 的边，生成树唯一，答案就是 w
```


#### 有向图的树形图

有向图用「入度矩阵 − 邻接矩阵」得到内向树（所有边指向根），去掉根所在的行列后求行列式。

```c++
ll arborescence(int n, vector<array<ll, 3>>& edges, int root, ll mod) {
    vector<vector<ll>> L(n, vector<ll>(n, 0));
    for (auto [u, v, w] : edges) {           // 边 u -> v
        L[v][v] = (L[v][v] + w) % mod;       // 入度
        L[v][u] = (L[v][u] - w % mod + mod) % mod;
    }
    // 去掉 root 所在的行和列
    vector<int> keep;
    for (int i = 0; i < n; i++) if (i != root) keep.push_back(i);
    int k = keep.size();
    vector<vector<ll>> A(k, vector<ll>(k));
    for (int i = 0; i < k; i++) for (int j = 0; j < k; j++) A[i][j] = L[keep[i]][keep[j]];
    return det_mod(A, k, mod);
}
// 外向树把入度换成出度（即交换 u、v 的角色）
```


## 快速傅里叶变换（FFT）


### 一句话

给定两个多项式 $A(x),B(x)$（次数 $\le 10^6$，整数系数），要你把它们乘起来并求出 $C(x)=A(x)B(x)$ 的每一项系数，系数大小可能撑满整型。**时间 $O(n\log n)$，空间 $O(n)$。**


### 原理


### 实现

（把系数补零到 $2$ 的幂，用单位根在「系数表示」与「点值表示」之间来回变换，点值相乘后再变回去。）

```c++
const double PI = acos(-1.0);
struct Cp {
    double x, y;
    Cp(double x = 0, double y = 0) : x(x), y(y) {}
    Cp operator+(const Cp& o) const { return {x + o.x, y + o.y}; }
    Cp operator-(const Cp& o) const { return {x - o.x, y - o.y}; }
    Cp operator*(const Cp& o) const { return {x * o.x - y * o.y, x * o.y + y * o.x}; }
    Cp operator/(double d) const { return {x / d, y / d}; }
};
void fft(vector<Cp>& a, int inv) {
    int n = a.size();
    for (int i = 1, j = 0; i < n; i++) {     // 位逆序置换
        int bit = n >> 1;
        for (; j & bit; bit >>= 1) j ^= bit;
        j ^= bit;
        if (i < j) swap(a[i], a[j]);
    }
    for (int len = 2; len <= n; len <<= 1) {
        double ang = 2 * PI / len * (inv ? -1 : 1);
        Cp w(cos(ang), sin(ang));
        for (int i = 0; i < n; i += len) {
            Cp cur(1, 0);
            for (int j = 0; j < len / 2; j++) {
                Cp u = a[i + j], v = cur * a[i + j + len / 2];
                a[i + j] = u + v, a[i + j + len / 2] = u - v;
                cur = cur * w;
            }
        }
    }
    if (inv) for (auto& x : a) x = x / n;
}
vector<ll> convolution_fft(vector<ll> A, vector<ll> B) {
    int need = A.size() + B.size() - 1;
    int n = 1;
    while (n < need) n <<= 1;
    vector<Cp> a(n), b(n);
    for (size_t i = 0; i < A.size(); i++) a[i] = Cp(A[i], 0);
    for (size_t i = 0; i < B.size(); i++) b[i] = Cp(B[i], 0);
    fft(a, 0), fft(b, 0);
    for (int i = 0; i < n; i++) a[i] = a[i] * b[i];
    fft(a, 1);
    vector<ll> res(need);
    for (int i = 0; i < need; i++) res[i] = llround(a[i].x);
    return res;
}
```


### 应用


#### 拆位处理大系数

系数到 $10^9$ 时 double 会丢精度：把一个系数拆成「高位 × 基数 + 低位」，做两次卷积再合并。

```c++
const int BASE = 32768;                      // 2^15，保证乘积不超 double 的精度
vector<ll> convolution_split(vector<ll> A, vector<ll> B) {
    vector<ll> ah(A.size()), al(A.size()), bh(B.size()), bl(B.size());
    for (size_t i = 0; i < A.size(); i++) ah[i] = A[i] / BASE, al[i] = A[i] % BASE;
    for (size_t i = 0; i < B.size(); i++) bh[i] = B[i] / BASE, bl[i] = B[i] % BASE;
    vector<ll> hh = convolution_fft(ah, bh);
    vector<ll> hl = convolution_fft(ah, bl);
    vector<ll> lh = convolution_fft(al, bh);
    vector<ll> ll_ = convolution_fft(al, bl);
    vector<ll> res(hh.size());
    for (size_t i = 0; i < res.size(); i++)
        res[i] = ((hh[i] % MOD) * BASE % MOD * BASE % MOD
                + ((hl[i] + lh[i]) % MOD) * BASE % MOD
                + ll_[i] % MOD) % MOD;
    return res;
}
// 三次 ~ 四次卷积，比单次慢但精度安全；模意义下更推荐直接上 NTT
```

#### 大整数乘法

把数字按位（或按 $10^4$ 进制）拆成多项式系数，乘完再统一处理进位。

```c++
string big_mul(const string& sa, const string& sb) {
    vector<ll> A, B;
    for (int i = sa.size() - 1; i >= 0; i--) A.push_back(sa[i] - '0');
    for (int i = sb.size() - 1; i >= 0; i--) B.push_back(sb[i] - '0');

    for (size_t i = 0; i + 1 < C.size(); i++) {   // 统一进位
        C[i + 1] += C[i] / 10;
        C[i] %= 10;
    }
    while (C.back() >= 10) {                      // 最高位可能还积着好几倍的进位
        ll t = C.back();
        C.back() = t % 10;
        C.push_back(t / 10);
    }
    while (C.size() > 1 && C.back() == 0) C.pop_back();
    string res;
    for (int i = C.size() - 1; i >= 0; i--) res += char('0' + C[i]);
    return res;
}
// 把进制从 10 换成 10^4 可以减少卷积长度，快好几倍
```

## 快速数论变换（NTT）


### 一句话

给定两个多项式（次数 $\le 10^6$，系数是模 $998244353$ 意义下的整数），要你求出它们的乘积的每一项系数，模数固定但卷积要反复做很多次。**时间 $O(n\log n)$，空间 $O(n)$。**


### 原理


### 实现

（把 FFT 的单位根换成模意义下的原根，全程整数运算，没有精度误差。）

```c++
const ll G = 3, MOD = 998244353;             // 998244353 = 119 * 2^23 + 1

ll qpow(ll a, ll b, ll mod) {
    ll r = 1;
    for (a %= mod; b; b >>= 1, a = (__int128)a * a % mod)
        if (b & 1) r = (__int128)r * a % mod;
    return r;
}
void ntt(vector<ll>& a, int inv) {
    int n = a.size();
    for (int i = 1, j = 0; i < n; i++) {
        int bit = n >> 1;
        for (; j & bit; bit >>= 1) j ^= bit;
        j ^= bit;
        if (i < j) swap(a[i], a[j]);
    }
    for (int len = 2; len <= n; len <<= 1) {
        ll w = qpow(G, (MOD - 1) / len, MOD);
        if (inv) w = qpow(w, MOD - 2, MOD);
        for (int i = 0; i < n; i += len) {
            ll cur = 1;
            for (int j = 0; j < len / 2; j++) {
                ll u = a[i + j];
                ll v = cur * a[i + j + len / 2] % MOD;
                a[i + j] = (u + v) % MOD;
                a[i + j + len / 2] = (u - v + MOD) % MOD;
                cur = cur * w % MOD;
            }
        }
    }
    if (inv) {
        ll invn = qpow(n, MOD - 2, MOD);
        for (auto& x : a) x = x * invn % MOD;
    }
}
vector<ll> convolution(vector<ll> A, vector<ll> B) {
    int need = A.size() + B.size() - 1;
    int n = 1;
    while (n < need) n <<= 1;
    A.resize(n), B.resize(n);
    ntt(A, 0), ntt(B, 0);
    for (int i = 0; i < n; i++) A[i] = A[i] * B[i] % MOD;
    ntt(A, 1);
    A.resize(need);
    return A;
}
```


### 应用


#### 任意模数 NTT（CRT 合并）

用三个 NTT 模数各做一次卷积，再用中国剩余定理合并，就能得到任意模数下的结果。

```c++
// 三个常用模数（都满足 2^23 | p-1）：998244353、1004535809、469762049
// 对每个模数跑一遍 NTT，得到 r1、r2、r3，再用 CRT 把它们拼回目标模数
// 数到 1e18 量级时要用 __int128；也可以先用三个模数还原真值再取模
const ll M1 = 998244353, M2 = 1004535809, M3 = 469762049;
ll crt3(ll r1, ll r2, ll r3, ll mod) {
    // 直接合并不变量：先算 r1 + M1 * t12，其中 t12 由 r2 决定
    ll t12 = (r2 - r1 % M2 + M2) % M2 * qpow(M1 % M2, M2 - 2, M2) % M2;
    __int128 x = (__int128)r1 + (__int128)M1 * t12;      // x 在 [0, M1*M2)
    ll t3 = (ll)((r3 - (ll)(x % M3) + M3) % M3 * qpow((ll)((__int128)M1 * M2 % M3), M3 - 2, M3) % M3);
    x += (__int128)M1 * M2 * t3;
    return (ll)(x % mod);
}
```

## 多项式运算


### 一句话

给定多项式 $A(x)$（次数 $<n$，$n\le 10^5$，模 $998244353$），要你依次求出它的逆、$\ln A$、$\exp A$、$\sqrt A$，以及 $A$ 除以 $B$ 的商与余数，同一个 $A$ 上这些运算会挨个问一遍。**时间 $O(n\log n)$，空间 $O(n)$。**


### 原理


### 实现

（核心是**多项式求逆**：牛顿迭代 $B\leftarrow B(2-AB)$，每次长度翻倍。）

```c++
// 前置：ntt、convolution、qpow（见上一节）
vector<ll> poly_inv(vector<ll> a, int n) {   // 要求 a[0] != 0
    vector<ll> b(1, qpow(a[0], MOD - 2, MOD));
    for (int m = 1; m < n; m <<= 1) {
        int sz = 1;
        while (sz < 4 * m) sz <<= 1;
        vector<ll> f(sz, 0), g(sz, 0);
        for (int i = 0; i < min((int)a.size(), 2 * m); i++) f[i] = a[i];
        for (int i = 0; i < (int)b.size(); i++) g[i] = b[i];
        ntt(f, 0), ntt(g, 0);
        for (int i = 0; i < sz; i++)              // b <- b * (2 - a * b)
            f[i] = g[i] * (2 - f[i] * g[i] % MOD + MOD) % MOD;
        ntt(f, 1);
        b.resize(min(n, 2 * m));
        for (int i = 0; i < (int)b.size(); i++) b[i] = f[i] % MOD;
    }
    b.resize(n);
    return b;
}
```


### 应用


#### 多项式求导与积分

求导：第 $i$ 项系数乘 $i+1$ 后左移一位；积分：第 $i$ 项系数除以 $i+1$ 后右移一位。两者都是 $O(n)$。

```c++
vector<ll> poly_deriv(vector<ll> a) {        // 求导
    if (a.size() <= 1) return {0};
    vector<ll> r(a.size() - 1);
    for (size_t i = 1; i < a.size(); i++) r[i - 1] = a[i] * i % MOD;
    return r;
}
vector<ll> poly_integ(vector<ll> a) {        // 积分（常数项取 0）
    vector<ll> r(a.size() + 1, 0);
    static vector<ll> invc;
    if ((int)invc.size() < (int)a.size() + 2) {
        invc.assign(a.size() + 2, 1);
        for (size_t i = 2; i < invc.size(); i++) invc[i] = (MOD - MOD / i) * invc[MOD % i] % MOD;
    }
    for (size_t i = 0; i < a.size(); i++) r[i + 1] = a[i] * invc[i + 1] % MOD;
    return r;
}
```


#### 多项式 $\ln$ 与快速幂

$\ln A=\int\frac{A'}{A}$（要求 $a_0=1$）；$A^k=\exp(k\ln A)$，$k$ 很大时要先对指数取模。

```c++
vector<ll> poly_ln(vector<ll> a, int n) {
    vector<ll> d = poly_deriv(a);
    vector<ll> inv = poly_inv(a, n);
    vector<ll> q = convolution(d, inv);
    q.resize(n);
    return poly_integ(q);
}
// 用法（k 可能是 long long）：
//   A^k = exp(k * ln A)，要求 a[0] == 1
//   若 a[0] != 1，先把 a[0]^k 提出来
```


#### 多项式 $\exp$ 与开方

$\exp$ 用牛顿迭代 $B\leftarrow B(1-\ln B+A)$；开方用 $B\leftarrow\frac{B+A/B}{2}$（要求 $a_0$ 是二次剩余）。

```c++
// exp 与 sqrt 的结构都是「牛顿迭代 + 每次调用一次 ln / inv」，
// 总复杂度 O(n log n)；这里给出可直接调用的外层骨架
vector<ll> poly_exp_skeleton(vector<ll> a, int n) {
    // 1. B <- [1]（a[0] 必须是 0）
    // 2. 每轮 m -> 2m：B <- B * (1 - ln(B) + A) 取前 2m 项
    vector<ll> b(1, 1);
    for (int m = 1; m < n; m <<= 1) {
        vector<ll> lb = poly_ln(b, 2 * m);     // 需要 b 补零到 2m
        lb.resize(2 * m, 0);
        for (int i = 0; i < 2 * m; i++) lb[i] = ((i < (int)a.size() ? a[i] : 0) - lb[i] + MOD) % MOD;
        lb[0] = (lb[0] + 1) % MOD;
        b = convolution(b, lb);
        b.resize(min(n, 2 * m));
    }
    b.resize(n);
    return b;
}
// sqrt A：b 从 a[0] 的平方根出发，每轮 b <- (b + A * inv(b)) / 2
// 除法取模：rev(A) = rev(Q) * rev(B) mod x^(n-m+1)，先求 rev(B) 的逆得到 Q，再算 R = A - Q * B
```


## 拉格朗日插值


### 一句话

给定 $n+1$ 个点 $(x_i,y_i)$（横坐标两两不同，$n\le 10^6$），要你找出过这些点的多项式并求出它在某个 $x_0$ 处的值，询问可能有很多次。**时间 $O(n)$（横坐标连续或等差时）或 $O(n\log^2 n)$（一般情况），空间 $O(n)$。**


### 原理


### 实现

（拉格朗日插值公式：$\displaystyle f(x)=\sum_{i=0}^{n}y_i\prod_{j\ne i}\frac{x-x_j}{x_i-x_j}$。）

```c++
vector<ll> y;
ll lagrange_consecutive(ll x, int n) {       // 点是 (0, y[0]), (1, y[1]), ..., (n-1, y[n-1])
    vector<ll> pre(n + 1, 1), suf(n + 1, 1);
    for (int i = 0; i < n; i++) pre[i + 1] = pre[i] * ((x - i) % MOD + MOD) % MOD;
    for (int i = n - 1; i >= 0; i--) suf[i] = suf[i + 1] * ((x - i) % MOD + MOD) % MOD;
    vector<ll> fac(n), ifac(n);
    fac[0] = 1;
    for (int i = 1; i < n; i++) fac[i] = fac[i - 1] * i % MOD;
    ifac[n - 1] = qpow(fac[n - 1], MOD - 2, MOD);
    for (int i = n - 1; i >= 1; i--) ifac[i - 1] = ifac[i] * i % MOD;
    ll ans = 0;
    for (int i = 0; i < n; i++) {
        ll t = y[i] * pre[i] % MOD * suf[i + 1] % MOD * ifac[i] % MOD * ifac[n - 1 - i] % MOD;
        if ((n - 1 - i) & 1) t = (MOD - t) % MOD;
        ans = (ans + t) % MOD;
    }
    return ans;
}
```


### 应用


#### 自然数幂和

$S_k(n)=\sum_{i=1}^{n}i^k$ 是 $n$ 的 $k+1$ 次多项式，插值 $k+2$ 个点即可；$k\le 10^5$ 时非常常用。

```c++
ll power_sum_lagrange(ll n, int k) {
    int m = k + 2;                            // 需要 m 个点
    y.assign(m, 0);
    ll acc = 0, p = 1;
    for (int i = 1; i <= m; i++) {            // 点 (i, sum_{j=1}^{i} j^k)
        p = 1;
        for (int t = 0; t < k; t++) p = p * i % MOD;
        acc = (acc + p) % MOD;
        y[i - 1] = acc;
    }
    if (n <= m) return y[n - 1];
    return lagrange_consecutive(n, m);
}
// 注意：这里是点 (1, ...)、(2, ...)，所以插值前要把 x 平移；简单起见可以
// 直接对「点 (1..m)」用差值版公式，或先求 g(n) = f(n+1) 的连续点插值
```


#### 横坐标是等差数列时的技巧

若 $x_i=a+id$，则 $x_i-x_j=(i-j)d$，于是分母 $\prod_{j\ne i}(x_i-x_j)=d^{n-1}\,i!\,(-1)^{n-1-i}(n-1-i)!$，整体仍是 $O(n)$。

```c++
ll lagrange_arith(vector<ll>& y, ll a, ll d, ll x, int n) {   // 点 (a+i*d, y[i])
    vector<ll> pre(n + 1, 1), suf(n + 1, 1), fac(n, 1), ifac(n, 1);
    for (int i = 0; i < n; i++) pre[i + 1] = pre[i] * ((x - a - (ll)i * d) % MOD + MOD) % MOD;
    for (int i = n - 1; i >= 0; i--) suf[i] = suf[i + 1] * ((x - a - (ll)i * d) % MOD + MOD) % MOD;
    for (int i = 1; i < n; i++) fac[i] = fac[i - 1] * i % MOD;
    ifac[n - 1] = qpow(fac[n - 1], MOD - 2, MOD);
    for (int i = n - 1; i >= 1; i--) ifac[i - 1] = ifac[i] * i % MOD;
    ll dn = qpow(d % MOD, n - 1, MOD), ans = 0;
    for (int i = 0; i < n; i++) {
        ll den = dn * fac[i] % MOD * fac[n - 1 - i] % MOD;
        if ((n - 1 - i) & 1) den = (MOD - den) % MOD;
        ans = (ans + y[i] * pre[i] % MOD * suf[i + 1] % MOD * qpow(den, MOD - 2, MOD)) % MOD;
    }
    return ans;
}
// 分母的逆元可以先用一次前缀积整体求出来再递推，把每个 i 的 log 也省掉
```


#### 判断「是不是多项式」

插值只能用于「答案确实是 $n$ 的多项式」的情形；先用暴力算前若干项、做差看第几阶差分变常数列。

```c++
bool is_polynomial(vector<ll> a, int maxd) { // a 是前若干项，检查 maxd 阶差分是否为常数

        if (same) return true;
        vector<ll> b;

        if (a.size() < 2) break;
    }

}
```

## 常系数齐次线性递推


### 一句话

给定 $k$ 阶常系数线性递推 $f_n=\sum_{i=1}^{k}c_if_{n-i}$ 与前 $k$ 项（$k\le 2000$），要你求出第 $n$ 项 $f_n$（$n\le 10^{18}$），$k$ 比矩阵做法能承受的更大而 $n$ 仍是天文数字。**时间 $O(k^2\log n)$，空间 $O(k)$。**


### 原理


### 实现

（Kitamasa：把递推写成特征多项式，用「多项式快速幂 + 多项式取模」把 $x^n$ 化简。）

```c++
// 在「模特征多项式」的意义下做乘法，O(k^2)
vector<ll> mulmod(const vector<ll>& a, const vector<ll>& b, const vector<ll>& c, int k) {
    vector<ll> t(2 * k - 1, 0);
    for (int i = 0; i < k; i++)
        for (int j = 0; j < k; j++)
            t[i + j] = (t[i + j] + a[i] * b[j]) % MOD;
    // 特征多项式：x^k = c1 x^(k-1) + ... + ck
    for (int i = 2 * k - 2; i >= k; i--)
        if (t[i])
            for (int j = 0; j < k; j++)
                t[i - 1 - j] = (t[i - 1 - j] + t[i] * c[j]) % MOD;
    return vector<ll>(t.begin(), t.begin() + k);
}
ll kitamasa(vector<ll> c, vector<ll> f, ll n) {   // c[i] 对应 c_{i+1}
    int k = c.size();
    if (n < (ll)f.size()) return f[n];
    vector<ll> a(k, 0), b(k, 0);
    a[0] = 1;
    if (k > 1) b[1] = 1;
    else b[0] = c[0];
    for (; n; n >>= 1, b = mulmod(b, b, c, k))
        if (n & 1) a = mulmod(a, b, c, k);
    ll ans = 0;
    for (int i = 0; i < k; i++) ans = (ans + a[i] * f[i]) % MOD;
    return ans;
}
```

```c++
    vector<ll> t(2 * k - 1, 0);

        for (int j = 0; j < k; j++)
            t[i + j] = (t[i + j] + a[i] * b[j]) % MOD;
    for (int i = 2 * k - 2; i >= k; i--)
```

### 应用


#### 打表 + BM + Kitamasa

只给了前若干项、系数未知时，先用 Berlekamp–Massey 求出最短递推式，再套 Kitamasa 求远处项。

```c++
vector<ll> bm(vector<ll> s) {
    vector<ll> c(1, 1), b(1, 1);
    int L = 0, m = 1;
    ll bd = 1;
    for (int n = 0; n < (int)s.size(); n++) {
        ll d = 0;
        for (int i = 0; i <= L; i++) d = (d + c[i] * s[n - i]) % MOD;
        if (!d) { m++; continue; }
        vector<ll> t = c;
        ll coef = d * qpow(bd, MOD - 2, MOD) % MOD;
        if ((int)c.size() < (int)b.size() + m) c.resize(b.size() + m);
        for (int i = 0; i < (int)b.size(); i++)
            c[i + m] = (c[i + m] - coef * b[i] % MOD + MOD) % MOD;
        if (2 * L <= n) { L = n + 1 - L, b = t, bd = d, m = 1; }
        else m++;
    }
    c.erase(c.begin());
    for (auto& x : c) x = (MOD - x) % MOD;
    return c;
}
// 用法：先暴力算前 2k 项，auto rc = bm(seq); 再 kitamasa(rc, seq, n)
```


#### 判定是否为线性递推

BM 返回的递推式拿去验算更多项：全部符合才说明是线性的，否则就是拟合过头了。

```c++
bool verify_bm(vector<ll>& seq, vector<ll> c) {
    int k = c.size();
    for (size_t i = k; i < seq.size(); i++) {
        ll t = 0;
        for (int j = 0; j < k; j++) t = (t + c[j] * seq[i - 1 - j]) % MOD;
        if (t != seq[i]) return false;
    }
    return true;
}
// 实际用法：打 200 项，BM 出递推式后拿剩下的 100 项验算
```


#### 快速求第 $n$ 项

见上面的 linear_rec；$k$ 阶递推的复杂度是 $O(k^2\log n)$，$k$ 到 $2000$ 时完全够用。

```c++
void check_linear_rec() {
    vector<ll> c = {1, 1}, f = {0, 1};
    // 与暴力递推对拍
    vector<ll> g(200, 0);
    g[0] = 0, g[1] = 1;
    for (int i = 2; i < 200; i++) g[i] = (g[i - 1] + g[i - 2]) % MOD;
    for (int i = 0; i < 200; i++) assert(linear_rec(c, f, i) == g[i]);
    assert(linear_rec(c, f, 10) == 55);
}
// k 阶递推的复杂度是 O(k^2 log n)；k 到 2000、n 到 1e18 大约 4e8 次运算，要卡常
```


#### Berlekamp–Massey 求递推式

只给了前若干项、系数未知时，用 BM 求出最短递推式，再套 linear_rec 求远处项。

```c++
vector<ll> bm_(vector<ll> s) {
    vector<ll> c(1, 1), b(1, 1);
    int L = 0, m = 1;
    ll bd = 1;
    for (int n = 0; n < (int)s.size(); n++) {
        ll d = 0;
        for (int i = 0; i <= L; i++) d = (d + c[i] * s[n - i]) % MOD;
        if (!d) { m++; continue; }
        vector<ll> t = c;
        ll coef = d * qpow(bd, MOD - 2, MOD) % MOD;
        if ((int)c.size() < (int)b.size() + m) c.resize(b.size() + m);
        for (int i = 0; i < (int)b.size(); i++)
            c[i + m] = (c[i + m] - coef * b[i] % MOD + MOD) % MOD;
        if (2 * L <= n) { L = n + 1 - L, b = t, bd = d, m = 1; }
        else m++;
    }
    c.erase(c.begin());
    for (auto& x : c) x = (MOD - x) % MOD;
    return c;
}
// 用法：先暴力算前 2k 项，auto rc = bm_(seq); 再 linear_rec(rc, seq, n)
```


#### 任意模数与多项式取模

模数不固定时把 mulmod_k 里的乘法换成 __int128；$k$ 很大（超过 $10^4$）时用 NTT 把多项式取模降到 $O(k\log k)$。

```c++
// 模数不固定时，把 mulmod_k 里的乘法换成 __int128 即可（下面的骨架展示替换点）
vector<ll> mulmod_any(const vector<ll>& a, const vector<ll>& b, const vector<ll>& c, int k, ll mod) {
    vector<ll> t(2 * k - 1, 0);
    for (int i = 0; i < k; i++)
        for (int j = 0; j < k; j++)
            t[i + j] = (ll)((t[i + j] + (__int128)a[i] * b[j]) % mod);   // 关键：__int128
    for (int i = 2 * k - 2; i >= k; i--)
        if (t[i])
            for (int j = 0; j < k; j++)
                t[i - 1 - j] = (ll)((t[i - 1 - j] + (__int128)t[i] * c[j]) % mod);
    return vector<ll>(t.begin(), t.begin() + k);
}
// 模数到 1e18 时，请特别小心「特征多项式的首位」——用「首一多项式」可以省掉一次求逆
```


## 快速沃尔什变换（FWT）


### 一句话

给定两个长度为 $2^k$ 的数组 $A,B$（$k\le 20$，元素模 $998244353$），要你对下标按位求出它们的或 / 与 / 异或卷积 $C_i=\sum_{j\oplus l=i}A_jB_l$，数组一次给全。**时间 $O(k2^k)$，空间 $O(2^k)$。**


### 原理


### 实现

```c++
const ll MOD = 998244353;
ll qpow(ll a, ll b, ll mod) {
    ll r = 1;
    for (a %= mod; b; b >>= 1, a = (__int128)a * a % mod)
        if (b & 1) r = (__int128)r * a % mod;
    return r;
}
void fwt_or(vector<ll>& a, int inv) {
    int n = a.size();
    for (int len = 1; len < n; len <<= 1)
        for (int i = 0; i < n; i += len << 1)
            for (int j = 0; j < len; j++) {
                if (!inv) a[i + j + len] = (a[i + j + len] + a[i + j]) % MOD;
                else a[i + j + len] = (a[i + j + len] - a[i + j] + MOD) % MOD;
            }
}
void fwt_and(vector<ll>& a, int inv) {
    int n = a.size();
    for (int len = 1; len < n; len <<= 1)
        for (int i = 0; i < n; i += len << 1)
            for (int j = 0; j < len; j++) {
                if (!inv) a[i + j] = (a[i + j] + a[i + j + len]) % MOD;
                else a[i + j] = (a[i + j] - a[i + j + len] + MOD) % MOD;
            }
}
void fwt_xor(vector<ll>& a, int inv) {
    int n = a.size();
    for (int len = 1; len < n; len <<= 1)
        for (int i = 0; i < n; i += len << 1)
            for (int j = 0; j < len; j++) {
                ll x = a[i + j], y = a[i + j + len];
                a[i + j] = (x + y) % MOD;
                a[i + j + len] = (x - y + MOD) % MOD;
            }
    if (inv) {
        ll invn = qpow(n, MOD - 2, MOD);
        for (auto& x : a) x = x * invn % MOD;
    }
}
```


### 应用


#### 子集或 / 与卷积

$c_k=\sum_{i\mid j=k}a_ib_j$ 用 or 卷积，$\sum_{i\&j=k}$ 用 and 卷积；长度必须是 $2$ 的幂。

```c++
vector<ll> conv_or(vector<ll> A, vector<ll> B) {
    fwt_or(A, 0), fwt_or(B, 0);
    for (size_t i = 0; i < A.size(); i++) A[i] = A[i] * B[i] % MOD;
    fwt_or(A, 1);
    return A;
}
vector<ll> conv_and(vector<ll> A, vector<ll> B) {
    fwt_and(A, 0), fwt_and(B, 0);
    for (size_t i = 0; i < A.size(); i++) A[i] = A[i] * B[i] % MOD;
    fwt_and(A, 1);
    return A;
}
```


#### 异或卷积

$c_k=\sum_{i\oplus j=k}a_ib_j$，与 or/and 的区别是逆变换要额外除以 $n$。

```c++
vector<ll> conv_xor(vector<ll> A, vector<ll> B) {
    fwt_xor(A, 0), fwt_xor(B, 0);
    for (size_t i = 0; i < A.size(); i++) A[i] = A[i] * B[i] % MOD;
    fwt_xor(A, 1);                           // 内部已除以 n
    return A;
}
```


#### 自卷积（求 $t$ 次幂）

要求「选 $t$ 个数的异或和为 $k$ 的方案数」时，先 FWT，再逐点取 $t$ 次幂，最后逆变换。

```c++
vector<ll> self_pow(vector<ll> A, ll t) {
    fwt_xor(A, 0);
    for (auto& x : A) x = qpow(x, t, MOD);   // 逐点幂
    fwt_xor(A, 1);
    return A;
}
// 例：A[S] 表示「选一个数得到 S」的方案数，self_pow(A, t) 就是「选 t 个数」
```


## 子集卷积


### 一句话

给定两个长度为 $2^k$ 的数组 $A,B$（$k\le 20$，元素模 $998244353$），要你对每个集合 $S$ 求出 $C_S=\sum_{T\subseteq S}A_T\,B_{S\setminus T}$，也就是下标取不相交并集的卷积，两个数组一次给全。**时间 $O(2^kk^2)$，空间 $O(2^kk)$。**


### 原理


### 实现

（把数组按 popcount 分层：第 $i$ 层只保留 $|S|=i$ 的元素，做 or 卷积后只取 $i+j=|S|$ 的项。）

```c++
// 前置：fwt_or（见上一节）
vector<ll> subset_conv(vector<ll> A, vector<ll> B, int k) {
    int n = 1 << k;
    vector<vector<ll>> fa(k + 1, vector<ll>(n, 0)), fb(k + 1, vector<ll>(n, 0));
    for (int S = 0; S < n; S++) {
        int c = __builtin_popcount(S);
        fa[c][S] = A[S], fb[c][S] = B[S];
    }
    for (int i = 0; i <= k; i++) fwt_or(fa[i], 0), fwt_or(fb[i], 0);
    vector<vector<ll>> fc(k + 1, vector<ll>(n, 0));
    for (int i = 0; i <= k; i++)
        for (int j = 0; i + j <= k; j++)
            for (int S = 0; S < n; S++)
                fc[i + j][S] = (fc[i + j][S] + fa[i][S] * fb[j][S]) % MOD;
    for (int i = 0; i <= k; i++) fwt_or(fc[i], 1);
    vector<ll> C(n, 0);
    for (int S = 0; S < n; S++) C[S] = fc[__builtin_popcount(S)][S];   // 只取「层数 = 大小」的
    return C;
}
```


### 应用


#### 不相交并集的计数

很多「把集合拆成两部分」的计数都靠它：令 A[T]=1 表示子集 T 合法，则「把全集拆成两个不相交合法子集」的方案数就是子集卷积在全集处的取值。

```c++
ll count_two_parts(vector<ll> A, int k, int full) {
    vector<ll> C = subset_conv(A, A, k);
    return C[full] % MOD;                    // 这里把「两份有序」都算了一次，不区分先后要再除以 2
}
// 例：k=2、A[{0}]=A[{1}]=1（其余为 0）时，full={0,1} 处得到 2（{0}|{1} 与 {1}|{0} 各算一次）
```


#### 恰好不相交的若干份

把子集卷积反复做 $t$ 次就能求「把 $S$ 拆成 $t$ 份互不相交的子集」的方案数；也可以对层数做多项式幂。

```c++
vector<ll> subset_conv_pow(vector<ll> A, int k, ll t) {
    // 每层单独做 t 次幂再组合（层数会相加，必须对齐）
    // 简化写法：反复调用 subset_conv 共 t-1 次
    vector<ll> r = A;
    for (ll i = 1; i < t; i++) r = subset_conv(r, A, k);
    return r;
}
// t 很大时改用「按层做多项式幂 + FWT」，复杂度 O(2^k k log k)
```


#### 与 SOS DP 的关系

只做 or 卷积、不要求「不相交」时，直接 SOS DP 更快（$O(k2^k)$）；两类结果在「存在有交集的下标对」时就会不同。

```c++
vector<ll> or_conv_only(vector<ll> A, vector<ll> B, int k) {
    int n = 1 << k;
    A.resize(n), B.resize(n);
    fwt_or(A, 0), fwt_or(B, 0);
    for (int i = 0; i < n; i++) A[i] = A[i] * B[i] % MOD;
    fwt_or(A, 1);
    return A;                                // 这是 or 卷积，不是子集卷积
}
bool differ_from_subset_conv(vector<ll> A, vector<ll> B, int k) {
    auto c1 = subset_conv(A, B, k), c2 = or_conv_only(A, B, k);
    for (size_t i = 0; i < c1.size(); i++) if (c1[i] != c2[i]) return true;
    return false;
}
// 取 A = B = [1,1,1,1]（k=2）时会返回 true：or 卷积把 {0,1}+{0,1} 这种也算进去了
```


## 高维前缀和（SOS DP）


### 一句话

给定一个长度为 $2^k$ 的数组 $A$（$k\le 20$），要你对每个集合 $S$ 求出它所有**子集**的和与所有**超集**的和，数组一次给全。**时间 $O(k2^k)$，空间 $O(2^k)$。**


### 原理


### 实现

```c++
const ll MOD = 998244353;
// ---------- 子集和：f[S] = sum_{T 是 S 的子集} a[T] ----------
void sos_subset(vector<ll>& f, int k) {
    int n = 1 << k;
    for (int i = 0; i < k; i++)
        for (int S = 0; S < n; S++)
            if (S >> i & 1)
                f[S] = (f[S] + f[S ^ (1 << i)]) % MOD;
}
// ---------- 超集和：g[S] = sum_{T 包含 S} a[T] ----------
void sos_superset(vector<ll>& g, int k) {
    int n = 1 << k;
    for (int i = 0; i < k; i++)
        for (int S = 0; S < n; S++)
            if (!(S >> i & 1))
                g[S] = (g[S] + g[S | (1 << i)]) % MOD;
}
```


### 应用


#### 逆变换（子集 / 超集差分）

把加法换成减法，顺序反过来，就得到「由子集和还原原数组」。

```c++
void sos_subset_inv(vector<ll>& f, int k) {  // f 是子集和，还原出原数组
    int n = 1 << k;
    for (int i = 0; i < k; i++)
        for (int S = 0; S < n; S++)
            if (S >> i & 1)
                f[S] = (f[S] - f[S ^ (1 << i)] + MOD) % MOD;
}
void sos_superset_inv(vector<ll>& g, int k) {
    int n = 1 << k;
    for (int i = 0; i < k; i++)
        for (int S = 0; S < n; S++)
            if (!(S >> i & 1))
                g[S] = (g[S] - g[S | (1 << i)] + MOD) % MOD;
}
// 两者互为逆运算：先 sos_subset 再 sos_subset_inv 应当还原
```


#### 与 FWT(or) 的关系

正向的 fwt_or 得到的就是「子集和」；所以 SOS DP 可以看成「只做正变换、不做逆变换」的 FWT。

```c++
bool check_sos_equals_fwt_or(vector<ll> a, int k) {
    vector<ll> x = a, y = a;
    sos_subset(x, k);
    fwt_or(y, 0);                            // 只做正变换
    for (size_t i = 0; i < x.size(); i++) if (x[i] != y[i]) return false;
    return true;                             // 对任意输入都应当相等
}
```


#### 统计 $\gcd$ 之类的高维前缀和

把「是 $d$ 的倍数」看成「下标是 $d$ 的超集」，就得到倍数方向的高维前缀和；有了它再配合莫比乌斯反演即可求「$\gcd$ 恰好为 $d$」。

```c++
vector<ll> multiple_count(vector<int>& a, int maxv) {   // g[d] = 有多少个数能被 d 整除
    vector<ll> g(maxv + 1, 0);
    for (int x : a) g[x]++;
    for (int d = 1; d <= maxv; d++)
        for (int m = 2 * d; m <= maxv; m += d) g[d] += g[m];
    return g;
}
// 复杂度 O(maxv log maxv)，是「倍数方向的 SOS」
// 想求「gcd 恰好为 d 的对数」时，用 mu 做一次反演：sum_{m 是 d 的倍数} mu(m/d) * C(g[m], 2)
```


#### 常见的「按位与为 0」计数

「有多少对 $(i,j)$ 使 $a_i\&a_j=0$」这类问题：先对 $a$ 做子集和，再对每个 $i$ 查 $a_i$ 的补集。

```c++
ll count_and_zero(vector<int>& a, int k) {
    int n = 1 << k;
    vector<ll> f(n, 0);
    for (int x : a) f[x]++;
    sos_subset(f, k);                        // f[S] = 值为 S 的子集的元素个数
    ll res = 0;
    int full = n - 1;
    for (int x : a) res += f[full ^ x];      // x 的补集（与 x 与运算为 0 的位）
    return res;
}
// 注意这里会把 i、j 的次序都算上；以及 x=0 时会把自己也算进去，按题意调整
```
