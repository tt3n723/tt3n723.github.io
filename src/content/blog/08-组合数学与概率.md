## 排列组合


### 一句话

给定 $n,m$（$0\le m\le n\le 10^7$）和素数模数 $p$（$p>n$），每次让你算出组合数 $\binom{n}{m}$ 与排列数 $P(n,m)=\frac{n!}{(n-m)!}$，两种询问交替出现。**时间 $O(n)$ 预处理、$O(1)$ 单次询问，空间 $O(n)$。**


### 原理


### 实现

```c++
const ll MOD = 998244353;
ll fac[N], ifac[N];

ll qpow(ll a, ll b, ll mod) {
    ll r = 1;
    for (a %= mod; b; b >>= 1, a = (__int128)a * a % mod)
        if (b & 1) r = (__int128)r * a % mod;
    return r;
}
void init_comb(int n) {
    fac[0] = 1;
    for (int i = 1; i <= n; i++) fac[i] = fac[i - 1] * i % MOD;
    ifac[n] = qpow(fac[n], MOD - 2, MOD);
    for (int i = n; i >= 1; i--) ifac[i - 1] = ifac[i] * i % MOD;
}
ll C(ll n, ll m) {
    if (m < 0 || m > n) return 0;
    return fac[n] * ifac[m] % MOD * ifac[n - m] % MOD;
}
ll P(ll n, ll m) {
    if (m < 0 || m > n) return 0;
    return fac[n] * ifac[n - m] % MOD;
}
```


### 应用


#### 二项式定理与杨辉三角

模数是合数、或者 $n$ 只有几千时，用不了费马小定理求逆元，直接用杨辉三角递推即可。

```c++
ll c[5005][5005];
void yanghui(int n, ll mod) {                // n 个元素以内都能算
    for (int i = 0; i <= n; i++) {
        c[i][0] = c[i][i] = 1 % mod;
        for (int j = 1; j < i; j++) c[i][j] = (c[i - 1][j - 1] + c[i - 1][j]) % mod;
    }
}
// 二项式定理：(a+b)^n = sum_k C(n,k) a^k b^(n-k)，也可以用它算系数
```


#### 隔板法与整数拆分

$x_1+\dots+x_k=n$ 且 $x_i\ge0$ 的方案数是 $\binom{n+k-1}{k-1}$；每个 $x_i$ 带上界时用容斥把「越界」的情形减掉。

```c++
ll bars_unbounded(ll n, int k) { return C(n + k - 1, k - 1); }

ll bars_bounded(ll n, int k, vector<ll>& cap) {   // 0 <= x_i <= cap[i]
    ll res = 0;
    for (int mask = 0; mask < (1 << k); mask++) {
        ll rem = n;
        int bits = __builtin_popcount(mask);
        for (int i = 0; i < k; i++) if (mask >> i & 1) rem -= cap[i] + 1;
        if (rem < 0) continue;
        ll t = C(rem + k - 1, k - 1);
        res = (res + (bits & 1 ? -t : t)) % MOD;
    }
    return (res % MOD + MOD) % MOD;
}
```


#### 可重组合与圆排列

从 $n$ 种元素里无序取 $m$ 个可重复的方案是 $\binom{n+m-1}{m}$；$n$ 个元素的圆排列是 $(n-1)!$。

```c++
ll multiset_choose(ll n, ll m) { return C(n + m - 1, m); }   // 可重组合
ll circular_perm(ll n) { return n <= 1 ? 1 : fac[n - 1]; }  // 圆排列
// 项链（旋转 + 翻转都算同一种）要用 Burnside，见后面的「Burnside 与 Pólya 计数」
```


#### 错排

$n$ 个元素的排列中「每个元素都不在自己原来位置」的个数：$D_n=(n-1)(D_{n-1}+D_{n-2})$。

```c++
ll derange[N];
void init_derange(int n) {
    derange[0] = 1, derange[1] = 0;
    for (int i = 2; i <= n; i++) derange[i] = (i - 1) * (derange[i - 1] + derange[i - 2]) % MOD;
}
// 通项：D_n = n! * sum_{i=0}^{n} (-1)^i / i!
```


## 鸽巢原理


### 一句话

给你一个长度为 $n$ 的整数序列 $a_1,\dots,a_n$（$n\le 10^6$，元素是任意整数），要你找出一个非空连续子段 $[l,r]$ 使它的和能被 $n$ 整除，并统计有多少个子段的和是 $n$ 的倍数，两种要求交替出现。**时间 $O(n)$，空间 $O(n)$。**


### 原理


### 实现

（看 $n+1$ 个前缀和 $S_0=0,S_1,\dots,S_n$ 对 $n$ 取模的余数：只有 $n$ 种取值却有 $n+1$ 个数，必有两个相同，它们的差就是一段和为 $n$ 的倍数的区间。）

```c++
int firstpos[N];
// 返回是否找到；找到时把区间写入 L、R
bool find_zero_sum_seg(int n, ll* a, int& L, int& R) {
    memset(firstpos, -1, sizeof(int) * (n + 1));
    firstpos[0] = 0;
    ll pre = 0;
    for (int i = 1; i <= n; i++) {
        pre = ((pre + a[i]) % n + n) % n;
        if (firstpos[pre] != -1) { L = firstpos[pre] + 1, R = i; return true; }
        firstpos[pre] = i;
    }
    return false;
}
```


### 应用


#### 和为 $n$ 的倍数的子段计数

同余的前缀和两两配对就能得到一段和为 $n$ 的倍数的区间；设余数 $c$ 出现了 $k$ 次，就贡献 $\binom{k}{2}$ 段。

```c++
ll count_zero_sum_seg(int n, ll* a) {
    vector<ll> cnt(n, 0);
    ll pre = 0;
    cnt[0] = 1;                              // 空前缀
    for (int i = 1; i <= n; i++) {
        pre = ((pre + a[i]) % n + n) % n;
        cnt[pre]++;
    }
    ll res = 0;
    for (ll k : cnt) res += k * (k - 1) / 2;
    return res;
}
```


#### 「两个数之差能被 $n$ 整除」

把「找区间」换成「找两个下标」，用的是同一条同余原理：$n+1$ 个数对 $n$ 取模必有重复。

```c++
// 给定 n+1 个整数，找出一对下标 (i, j) 使 a[i] - a[j] 能被 n 整除
pair<int, int> diff_multiple(int n, ll* a) {
    vector<int> pos(n, -1);
    for (int i = 1; i <= n + 1; i++) {
        int r = ((a[i] % n) + n) % n;
        if (pos[r] != -1) return {pos[r], i};
        pos[r] = i;
    }
    return {-1, -1};
}
// 鸽巢的一般形式：n 个物品放进 k 个盒子，必有一个盒子至少 ceil(n/k) 个
```


## 容斥原理


### 一句话

给定 $N$（$N\le 10^{18}$）和 $k$ 个数 $p_1,\dots,p_k$（$k\le 20$，$p_i\le 10^9$），要你数出 $[1,N]$ 中至少被某个 $p_i$ 整除的数的个数。**时间 $O(2^k\log p)$，空间 $O(1)$。**


### 原理


### 实现

```c++
ll gcd(ll a, ll b) { return b ? gcd(b, a % b) : a; }

// 求 [1, N] 中至少被某个 p[i] 整除的个数
ll inclusion_exclusion(ll N, vector<ll>& p) {
    int k = p.size();
    ll res = 0;
    for (int mask = 1; mask < (1 << k); mask++) {
        ll lcm = 1;
        int bits = __builtin_popcount(mask);
        for (int i = 0; i < k; i++)
            if (mask >> i & 1) {
                lcm = lcm / gcd(lcm, p[i]) * p[i];
                if (lcm > N) break;          // 已经超过 N，这个集合并集为空
            }
        if (lcm > N) continue;
        res += (bits & 1 ? 1 : -1) * (N / lcm);
    }
    return res;
}
```


### 应用


#### 与 $M$ 互质的数的个数

把 $M$ 分解质因数，容斥「至少被某个质因子整除」的个数，再用 $N$ 减掉。

```c++
ll coprime_count(ll N, ll M) {
    vector<ll> p;
    for (ll i = 2; i * i <= M; i++)
        if (M % i == 0) { p.push_back(i); while (M % i == 0) M /= i; }
    if (M > 1) p.push_back(M);
    return N - inclusion_exclusion(N, p);
}
```


#### 互质数对计数（莫比乌斯容斥）

$$\sum_{i\le n}\sum_{j\le m}[\gcd(i,j)=1]=\sum_{d}\mu(d)\left\lfloor\frac nd\right\rfloor\left\lfloor\frac md\right\rfloor,$$
再用整除分块把求和压到 $O(\sqrt{\min(n,m)})$。

```c++
int mu[N], prime[N], pcnt;
bool vis[N];
ll premu[N];
void sieve_mu(int n) {                       // 线性筛莫比乌斯函数
    mu[1] = 1;
    for (int i = 2; i <= n; i++) {
        if (!vis[i]) prime[++pcnt] = i, mu[i] = -1;
        for (int j = 1; j <= pcnt && (ll)i * prime[j] <= n; j++) {
            int x = i * prime[j];
            vis[x] = true;
            if (i % prime[j] == 0) { mu[x] = 0; break; }
            mu[x] = -mu[i];
        }
    }
    for (int i = 1; i <= n; i++) premu[i] = premu[i - 1] + mu[i];
}
ll coprime_pairs(int n, int m) {
    sieve_mu(min(n, m));
    ll res = 0, lim = min(n, m);
    for (ll l = 1, r; l <= lim; l = r + 1) {
        r = min(n / (n / l), m / (m / l));
        res += (premu[r] - premu[l - 1]) * (n / l) * (m / l);
    }
    return res;
}
```


#### 错排的容斥推导

$\sum_k(-1)^k\binom nk(n-k)!$：先任排，再容斥「某个元素在自己的位置上」。可以用它验证递推式。

```c++
ll derange_by_ie(int n) {
    ll res = 0;
    for (int k = 0; k <= n; k++) {
        ll t = C(n, k) * fac[n - k] % MOD;
        res = (res + (k & 1 ? -t : t)) % MOD;
    }
    return (res % MOD + MOD) % MOD;
}
```


## 卡特兰数


### 一句话

给定 $n$（$0\le n\le 10^7$）和素数模数 $p$（$p>2n$），要你算出第 $n$ 个卡特兰数 $\mathrm{Cat}_n=\frac{1}{n+1}\binom{2n}{n}$。**时间 $O(n)$ 预处理后 $O(1)$ 单次，空间 $O(n)$。**


### 原理


### 实现

```c++
const ll MOD = 998244353;
ll fac[N], ifac[N];
ll qpow(ll a, ll b, ll mod) {
    ll r = 1;
    for (a %= mod; b; b >>= 1, a = (__int128)a * a % mod)
        if (b & 1) r = (__int128)r * a % mod;
    return r;
}
void init_comb(int n) {
    fac[0] = 1;
    for (int i = 1; i <= n; i++) fac[i] = fac[i - 1] * i % MOD;
    ifac[n] = qpow(fac[n], MOD - 2, MOD);
    for (int i = n; i >= 1; i--) ifac[i - 1] = ifac[i] * i % MOD;
}
ll C(ll n, ll m) { return (m < 0 || m > n) ? 0 : fac[n] * ifac[m] % MOD * ifac[n - m] % MOD; }

// ---------- 通项：Cat_n = C(2n, n) / (n + 1) ----------
ll catalan(int n) { return C(2 * n, n) * qpow(n + 1, MOD - 2, MOD) % MOD; }

// ---------- 递推：Cat_0 = 1，Cat_n = sum_{i=0}^{n-1} Cat_i * Cat_{n-1-i} ----------
ll cat[N];
void catalan_dp(int n) {
    cat[0] = 1;
    for (int i = 1; i <= n; i++) {
        cat[i] = 0;
        for (int j = 0; j < i; j++) cat[i] = (cat[i] + cat[j] * cat[i - 1 - j]) % MOD;
    }
}
```


### 应用


#### 合法括号序列的判定与计数

$n$ 对括号的合法序列个数就是 $\mathrm{Cat}_n$；判定一个给定序列是否合法只要栈模拟。

```c++
bool valid_bracket(const string& s) {
    int bal = 0;
    for (char c : s) {
        bal += (c == '(' ? 1 : -1);
        if (bal < 0) return false;            // 任何前缀都不能右括号更多
    }
    return bal == 0;
}
```


#### 字典序第 $k$ 个合法括号序列

用 $f[i][j]$（还剩 $i$ 个左括号、$j$ 个右括号时的合法填法数）逐位「试填左括号」，数不够就减掉并改填右括号。$f$ 满足 $f[i][j]=f[i-1][j]+f[i][j-1]$。

```c++
string kth_bracket(int n, ll k) {
    // g[a][b]：已放 a 个左括号、b 个右括号，走到 (n,n) 且全程 a' >= b' 的方案数
    vector<vector<ll>> g(n + 1, vector<ll>(n + 1, 0));
    g[n][n] = 1;
    for (int a = n; a >= 0; a--)
        for (int b = n; b >= 0; b--) {
            if (a == n && b == n) continue;
            if (b > a) { g[a][b] = 0; continue; }
            ll t = 0;
            if (a + 1 <= n) t += g[a + 1][b];          // 放一个左括号
            if (b + 1 <= n && b + 1 <= a) t += g[a][b + 1];   // 放一个右括号（要求放完仍 a >= b）
            g[a][b] = t;
        }
    if (n == 0) return "";
    string res;
    int a = 0, b = 0;
    while (a < n || b < n) {
        if (a < n && g[a + 1][b] >= k) { res += '('; a++; }   // 字典序上 '(' 更小，先试它
        else {
            if (a < n) k -= g[a + 1][b];
            res += ')';
            b++;
        }
    }
    return res;
}
```


#### 凸多边形三角剖分数

$m$ 个顶点的凸多边形的三角剖分数是 $\mathrm{Cat}_{m-2}$，用区间 DP 也能算出来（$f[i][j]=\sum_k f[i][k]f[k][j]$）。

```c++
ll triangulation(int m) {                    // m 个顶点，m >= 2
    vector<vector<ll>> f(m, vector<ll>(m, 0));
    for (int i = 0; i + 1 < m; i++) f[i][i + 1] = 1;
    for (int len = 2; len < m; len++)
        for (int i = 0; i + len < m; i++) {
            int j = i + len;
            for (int k = i + 1; k < j; k++)
                f[i][j] = (f[i][j] + f[i][k] * f[k][j]) % MOD;
        }
    return f[0][m - 1];
}
// 结论：triangulation(m) == Cat_{m-2}
```


#### 出栈序列的合法性

$n$ 个元素 $1..n$ 依次入栈，判断给定的出栈序列是否可行；可行的序列总数正是 $\mathrm{Cat}_n$。

```c++
bool valid_pop(int n, const vector<int>& b) {
    vector<int> st;
    int nxt = 1;
    for (int x : b) {
        while ((st.empty() || st.back() != x) && nxt <= n) st.push_back(nxt++);
        if (st.empty() || st.back() != x) return false;
        st.pop_back();
    }
    return true;
}
```


## 斯特林数


### 一句话

给定 $n,k$（$0\le k\le n\le 5000$），要你算出第二类斯特林数 $S(n,k)$（$n$ 个不同元素划分成 $k$ 个非空子集的方案数）与第一类斯特林数 $s(n,k)$（$n$ 个元素排成 $k$ 个循环的方案数），两种询问交替出现。**时间 $O(nk)$，空间 $O(nk)$。**


### 原理


### 实现

```c++
const ll MOD = 998244353;
ll S2[N][N], S1[N][N];

// ---------- 第二类：S(n,k) = S(n-1,k-1) + k * S(n-1,k) ----------
void init_stirling2(int n) {
    S2[0][0] = 1;
    for (int i = 1; i <= n; i++)
        for (int j = 1; j <= i; j++)
            S2[i][j] = (S2[i - 1][j - 1] + j * S2[i - 1][j]) % MOD;
}
// ---------- 第一类（无符号）：s(n,k) = s(n-1,k-1) + (n-1) * s(n-1,k) ----------
void init_stirling1(int n) {
    S1[0][0] = 1;
    for (int i = 1; i <= n; i++)
        for (int j = 1; j <= i; j++)
            S1[i][j] = (S1[i - 1][j - 1] + (i - 1) * S1[i - 1][j]) % MOD;
}
```


### 应用


#### 幂与下降幂的转换

$x^n=\sum_k S(n,k)\,x^{\underline{k}}$，其中 $x^{\underline{k}}=x(x-1)\cdots(x-k+1)$。这条恒等式是第二类斯特林数最重要的用途。

```c++
ll falling_pow(ll x, int k) {
    ll r = 1;
    for (int i = 0; i < k; i++) r = r * ((x - i) % MOD + MOD) % MOD;
    return r;
}
ll power_via_stirling(ll x, int n) {         // 算出 x^n
    init_stirling2(n);
    ll res = 0;
    for (int k = 0; k <= n; k++) res = (res + S2[n][k] * falling_pow(x, k)) % MOD;
    return res;
}
```


#### 第一类与上升幂

$x^{\overline{n}}=x(x+1)\cdots(x+n-1)=\sum_k s(n,k)\,x^k$。

```c++
ll rising_pow_expand(ll x, int n) {          // 展开 x 的上升幂
    init_stirling1(n);
    ll res = 0, p = 1;
    for (int k = 0; k <= n; k++) {
        res = (res + S1[n][k] * p) % MOD;
        p = p * (x % MOD) % MOD;
    }
    return res;
}
```


#### 自然数幂和

$$\sum_{i=0}^{n}i^k=\sum_{j=0}^{k}S(k,j)\cdot j!\cdot\binom{n+1}{j+1},$$
配合「$O(k)$ 算一个 $\binom{n+1}{j+1}$」就能在 $O(k^2)$ 内回答一次幂和。注意左边是从 $i=0$ 开始的：$k\ge1$ 时它和 $\sum_{i=1}^{n}i^k$ 相同，$k=0$ 时右边要多减 $1$。

```c++
ll C_big_small_m(ll n, ll m) {               // m 很小、n 可以很大
    if (m < 0 || m > n) return 0;
    ll a = 1, b = 1;
    for (ll i = 0; i < m; i++) {
        a = a * ((n - i) % MOD) % MOD;
        b = b * (i + 1) % MOD;
    }
    return a * qpow(b, MOD - 2, MOD) % MOD;
}
ll power_sum(ll n, int k) {                  // sum_{i=1}^{n} i^k
    init_stirling2(k);
    ll res = 0, fact = 1;
    for (int j = 0; j <= k; j++) {
        if (j) fact = fact * j % MOD;
        res = (res + S2[k][j] * fact % MOD * C_big_small_m(n + 1, j + 1)) % MOD;
    }
    return res;
}
```


#### 第二类斯特林数的通项

$S(n,k)=\frac{1}{k!}\sum_{i=0}^{k}(-1)^i\binom{k}{i}(k-i)^n$：$k$ 不大而 $n$ 很大时用这个，$O(k\log n)$ 一次。

```c++
ll qpow_(ll a, ll b, ll mod) {
    ll r = 1;
    for (a %= mod; b; b >>= 1, a = (__int128)a * a % mod)
        if (b & 1) r = (__int128)r * a % mod;
    return r;
}
ll stirling2_formula(ll n, int k) {
    // 这里用 O(k) 的乘积式算 C(k, i) 与 1/k!，不依赖外部的阶乘表
    ll res = 0, cki = 1, invk = 1;
    for (int i = 0; i <= k; i++) {
        ll t = cki * qpow_(k - i, n, MOD) % MOD;
        res = (res + (i & 1 ? -t : t)) % MOD;
        cki = cki * (k - i) % MOD * qpow_(i + 1, MOD - 2, MOD) % MOD;   // C(k, i+1)
    }
    for (int i = 1; i <= k; i++) invk = invk * i % MOD;
    return (res % MOD + MOD) % MOD * qpow_(invk, MOD - 2, MOD) % MOD;
}
```


## 贝尔数


### 一句话

给定 $n$（$0\le n\le 5000$），要你算出贝尔数 $B_n$（$n$ 个不同元素划分成任意个非空子集的方案数）。**时间 $O(n^2)$，空间 $O(n^2)$。**


### 原理


### 实现

```c++
const ll MOD = 998244353;
ll C[5005][5005], B[N], tri[1005][1005];

// ---------- 递推：B_{n+1} = sum_{k=0}^{n} C(n, k) * B_k ----------
void bell_dp(int n) {
    for (int i = 0; i <= n; i++) {
        C[i][0] = C[i][i] = 1;
        for (int j = 1; j < i; j++) C[i][j] = (C[i - 1][j - 1] + C[i - 1][j]) % MOD;
    }
    B[0] = 1;
    for (int i = 0; i < n; i++) {
        ll s = 0;
        for (int k = 0; k <= i; k++) s = (s + C[i][k] * B[k]) % MOD;
        B[i + 1] = s;
    }
}
// ---------- 贝尔三角形：每行首位 = 上一行末位，其余 = 左上方 + 左方 ----------
void bell_triangle(int n) {
    tri[0][0] = 1, B[0] = 1;
    for (int i = 1; i <= n; i++) {
        tri[i][0] = tri[i - 1][i - 1];
        for (int j = 1; j <= i; j++) tri[i][j] = (tri[i][j - 1] + tri[i - 1][j - 1]) % MOD;
        B[i] = tri[i][0];
```

#### 集合划分总数（第二类斯特林数按行求和）

$B_n=\sum_{k=0}^{n}S(n,k)$：先算一行的斯特林数再求和，是最直观的算法。

```c++
ll bell_by_stirling(int n) {
    vector<ll> S(n + 1, 0);
    S[0] = 1;                                 // 只保留「上一个 i」这一行，空间 O(n)
    for (int i = 1; i <= n; i++) {
        vector<ll> T(n + 1, 0);
        for (int j = 1; j <= i; j++) T[j] = (S[j - 1] + j * S[j]) % MOD;
        S.swap(T);
    }
    ll res = 0;
    for (int j = 0; j <= n; j++) res = (res + S[j]) % MOD;
    return res;
}
```

#### 贝尔数对任意模数的计算

```c++

    vector<vector<ll>> C(n + 1, vector<ll>(n + 1, 0));
    for (int i = 0; i <= n; i++) {
        C[i][0] = C[i][i] = 1 % mod;
        for (int j = 1; j < i; j++) C[i][j] = (C[i - 1][j - 1] + C[i - 1][j]) % mod;

        for (int k = 0; k <= i; k++) s = (s + C[i][k] * B[k]) % mod;

    return B[n];
}
```

## 生成函数


### 一句话

给定一个计数问题（规模 $n\le 5000$，模 $998244353$），要你用普通生成函数（OGF）或指数生成函数（EGF）把答案的系数算出来，比如「把 $n$ 拆成正整数之和（不计顺序）的方案数」或从递推的生成函数求通项，几种问题交替出现。**时间 $O(n^2)$，空间 $O(n)$。**


### 原理


### 实现

```c++

ll qpow(ll a, ll b, ll mod) {
    ll r = 1;
    for (a %= mod; b; b >>= 1, a = (__int128)a * a % mod)
        if (b & 1) r = (__int128)r * a % mod;
    return r;
}
// ---------- 整数拆分的 OGF：prod_{i>=1} 1 / (1 - x^i) ----------
// 直接按「完全背包」展开这个无穷乘积，O(n^2)
vector<ll> partition(int n) {
    vector<ll> f(n + 1, 0);
    f[0] = 1;
    for (int i = 1; i <= n; i++)
        for (int j = i; j <= n; j++)
            f[j] = (f[j] + f[j - i]) % MOD;
    return f;
}
// ---------- 朴素多项式求逆（OGF 里最常用的一个操作）：O(n^2) ----------
// B = 1 / A，用牛顿迭代的朴素版本 B <- B * (2 - A * B)
vector<ll> poly_inv_naive(vector<ll> a, int n) {
    vector<ll> b(n, 0);
    b[0] = qpow(a[0], MOD - 2, MOD);
    for (int i = 1; i < n; i++) {
        ll s = 0;
        for (int j = 1; j <= i; j++) s = (s + a[j] * b[i - j]) % MOD;
        b[i] = (MOD - s) * b[0] % MOD;
    }
    return b;
}
```

#### 范德蒙德卷积

#### 拉格朗日反演

若 $F(0)=0$ 且 $F'(0)\ne0$，$G$ 是 $F$ 的复合逆，则
$$[x^n]G(x)=\frac1n\,[x^{n-1}]\left(\frac{x}{F(x)}\right)^n .$$
用它可以一步推出卡特兰数等通项；下面是「用已知的卡特兰 OGF 方程反解系数」的验证写法。

```c++
// 卡特兰数的 OGF 满足 C(x) = 1 + x*C(x)^2；令 F = C - 1，则 F = x(1+F)^2，
// 复合逆 G = x/(1+x)^2，由拉格朗日反演：
//   [x^n]F = (1/n) * [x^(n-1)] (1+x)^(2n) = (1/n) * C(2n, n-1)
// 而 C(2n, n-1)/n 恰好等于 C(2n, n)/(n+1) = Cat_n
ll catalan_by_lagrange(int n) {
    if (n == 0) return 1;
    ll a = 1, b = 1;
    for (int i = 0; i < n - 1; i++) { a = a * ((2 * n - i) % MOD) % MOD; b = b * (i + 1) % MOD; }
    return a * qpow(b, MOD - 2, MOD) % MOD * qpow(n, MOD - 2, MOD) % MOD;
}
// 验证：catalan_by_lagrange(n) 与 C(2n,n)/(n+1) 相等
```


#### 指数生成函数（EGF）与集合的划分

排列类问题用 EGF：$A(x)=\sum a_n\frac{x^n}{n!}$，卷积前要除以阶乘。例如「$n$ 个元素分成若干非空有标号集合」的 EGF 是 $e^{e^x-1}$，系数乘 $n!$ 就是贝尔数。

```c++
// 只用 OGF 手段也能求贝尔数：B_n = sum_k S(n,k)
// 下面用 EGF 的思路做「n 个位置的错排」：
//   D(x) = e^{-x} / (1 - x)，系数乘 n! 就是错排数
vector<ll> derange_by_egf(int n) {
    vector<ll> fac(n, 1), ifac(n, 1);
    for (int i = 1; i < n; i++) fac[i] = fac[i - 1] * i % MOD;
    ifac[n - 1] = qpow(fac[n - 1], MOD - 2, MOD);
    for (int i = n - 1; i >= 1; i--) ifac[i - 1] = ifac[i] * i % MOD;

    vector<ll> ex(n), fx(n, 1), d(n, 0);
    for (int i = 0; i < n; i++) ex[i] = (i % 2 ? MOD - 1 : 1) * ifac[i] % MOD;   // e^{-x}
    for (int i = 0; i < n; i++)
        for (int j = 0; i + j < n; j++)
            d[i + j] = (d[i + j] + ex[i] * fx[j]) % MOD;     // 与 1/(1-x) 卷积
    for (int i = 0; i < n; i++) d[i] = d[i] * fac[i] % MOD;  // EGF 的系数要乘回 n!
    return d;
}
```

#### Prüfer 序列的双向构造

树 → 序列：每次取编号最小的叶子，把它的邻居写进序列并删掉它，重复 $n-2$ 次。序列 → 树：按「出现次数 = 度数 − 1」用优先队列反向还原。

```c++
// 树 -> Prüfer 序列（vertex 编号 1..n）
vector<int> to_prufer(int n, vector<int>& deg, vector<vector<int>>& g) {
    priority_queue<int, vector<int>, greater<int>> leaf;
    for (int i = 1; i <= n; i++) if (deg[i] == 1) leaf.push(i);
    vector<int> seq;
    for (int t = 1; t <= n - 2; t++) {
        int u = leaf.top(); leaf.pop();
        int v = -1;
        for (int x : g[u]) if (deg[x] > 0) { v = x; break; }
        seq.push_back(v);
        if (--deg[v] == 1) leaf.push(v);
        deg[u] = 0;
    }
    return seq;
}
// 序列 -> 树：把每个点的出现次数 + 1 当作度数，同样用最小堆配对还原
```

## 竞赛图计数

### 一句话

给定 $n$（$n\le 10^5$）与一个出度序列，要你数出出度恰好是这个序列的竞赛图个数。**时间 $O(n^2)$ 或 $O(n\log n)$，空间 $O(n)$。**

### 原理


### 实现

（竞赛图的「凝聚」是一个全序：任意两个强连通分量之间的边方向一致。于是按「拓扑序最靠前的那个强连通分量」分类，就能写出递推。）

```c++
// 强连通竞赛图计数
//   总方案数 = 2^{C(n,2)}（每对点任选方向）
//   减去「最小分量大小为 k」的情形：C(n,k) * f[k] * 2^{C(n-k,2)}
//   （前 k 个点构成一个强连通分量，剩下 n-k 个点之间的边可以任意，
//     而两组之间的边方向被「前 k 个点在拓扑序最前」唯一确定）
//   f[n] = 2^{C(n,2)} - sum_{k=1}^{n-1} C(n,k) * f[k] * 2^{C(n-k,2)}
ll f[N], C[N][N], pw2[N * N / 2 + 5];
void tournament_count(int n, ll MOD) {
    for (int i = 0; i <= n; i++) {
        C[i][0] = 1;
        for (int j = 1; j <= i; j++) C[i][j] = (C[i - 1][j - 1] + C[i - 1][j]) % MOD;
    }
    pw2[0] = 1;
    for (int i = 1; i <= n * (n - 1) / 2; i++) pw2[i] = pw2[i - 1] * 2 % MOD;
    for (int m = 1; m <= n; m++) {
        ll s = pw2[m * (m - 1) / 2];
        for (int k = 1; k < m; k++)
            s = (s - C[m][k] * f[k] % MOD * pw2[(m - k) * (m - k - 1) / 2]) % MOD;
        f[m] = (s % MOD + MOD) % MOD;
    }
}
// 例：f[1] = 1，f[2] = 0（两个点的竞赛图一定不强连通），f[3] = 2（两个三元环）。
// 验证 f[3]：2^3 - C(3,1)*f[1]*2^1 - C(3,2)*f[2]*2^0 = 8 - 6 - 0 = 2 ✓
```

```c++
vector<double> solve_linear(vector<vector<double>> A, vector<double> b, int n) {
    for (int c = 0; c < n; c++) {
        int p = c;
        for (int i = c + 1; i < n; i++) if (fabs(A[i][c]) > fabs(A[p][c])) p = i;
        swap(A[c], A[p]), swap(b[c], b[p]);
        double inv = 1.0 / A[c][c];
        for (int j = c; j < n; j++) A[c][j] *= inv;
        b[c] *= inv;
        for (int i = 0; i < n; i++) {
            if (i == c) continue;
            double t = A[i][c];
            for (int j = c; j < n; j++) A[i][j] -= t * A[c][j];
            b[i] -= t * b[c];
        }
    }
    return b;                                // b[i] 就是 E_i
}
// 例：0..n 这些格子，走到 0 或 n 就停，E[i] = 1 + 0.5 E[i-1] + 0.5 E[i+1]
//   => E[i] - 0.5 E[i-1] - 0.5 E[i+1] = 1，边界 E[0] = E[n] = 0
```

#### 条件概率与贝叶斯

$P(A\mid B)=\frac{P(A\cap B)}{P(B)}$，贝叶斯公式 $P(A\mid B)=\frac{P(B\mid A)P(A)}{P(B)}$；全概率公式 $P(A)=\sum_i P(A\mid B_i)P(B_i)$。

```c++
// 例：三扇门问题。换了之后选中的概率
double monty_hall_switch() {
    double prior = 1.0 / 3;                  // 一开始选中
    // 主持人总是打开一扇空门，换门后选中的概率 = 1 - 1/3
    return 1 - prior;
}
```

### 一句话

### 实现

```c++
// ---------- Nim：异或和非零则先手必胜 ----------
bool nim_win(int* a, int n) {
    int s = 0;
    for (int i = 1; i <= n; i++) s ^= a[i];
    return s != 0;
}
// ---------- 反 Nim（取走最后一颗的人输） ----------
bool misere_nim(int* a, int n) {
    int s = 0;
    bool allone = true;
    for (int i = 1; i <= n; i++) {
        s ^= a[i];
        if (a[i] > 1) allone = false;
    }
    return allone ? s == 0 : s != 0;
}
// ---------- SG 函数：sg(u) = mex{ sg(v) : u -> v } ----------
int sg[N];
vector<int> g[N];
int get_sg(int u) {
    if (sg[u] != -1) return sg[u];
    bool seen[N] = {};
    for (int v : g[u]) seen[get_sg(v)] = true;
    int m = 0;
    while (seen[m]) m++;
    return sg[u] = m;
}
```

#### 多个游戏之和

总 SG 等于各子游戏 SG 的异或和，非零先手必胜；这就是 Nim 的推广。

```c++
}
```

#### 阶梯 Nim

石子放在台阶 $1..n$ 上，每次把某一级的若干石子移到下一级，移到第 $0$ 级后不能再动；等价于只看**奇数编号台阶**上的 Nim。

```c++
bool stair_nim(vector<int>& a, int n) {      // a[1..n]，a[0] 是地面
    int s = 0;
    for (int i = n; i >= 1; i -= 2) s ^= a[i];
    return s != 0;
}
```


#### 威佐夫博弈

两堆石子，每次可以从一堆取任意多、或从两堆取同样多；必败态是 $(\lfloor k\phi\rfloor,\lfloor k\phi^2\rfloor)$。

```c++
bool wythoff_lose(ll a, ll b) {
    if (a > b) swap(a, b);
    double phi = (1 + sqrt(5.0)) / 2;
    ll k = (ll)((b - a) * phi);
    return a == k;                            // 命中必败态
}
```
