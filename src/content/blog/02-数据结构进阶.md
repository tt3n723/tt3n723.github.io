## 并查集


### 一句话

一开始有 $n$ 个互不相通的点和 $m$ 次操作（$n,m\le 10^6$），每次操作是「在 $a$ 与 $b$ 之间加一条边」或「询问 $a$ 与 $b$ 现在是否连通」，两种操作交替出现。**时间均摊 $O(\alpha(n))$ 每次，空间 $O(n)$。**


### 原理


### 实现

```c++
int fa[N], sz[N], comps;                     // comps：连通块个数

void init(int n) {
    for (int i = 1; i <= n; i++) fa[i] = i, sz[i] = 1;
    comps = n;
}
int find(int x) {
    while (fa[x] != x) x = fa[x] = fa[fa[x]];   // 路径压缩（迭代版，防爆栈）
    return x;
}
bool unite(int a, int b) {
    int x = find(a), y = find(b);
    if (x == y) return false;
    if (sz[x] < sz[y]) swap(x, y);           // 按大小合并
    fa[y] = x, sz[x] += sz[y], comps--;
    return true;
}
bool same(int a, int b) { return find(a) == find(b); }
```


合并时把大小相加、把连通块计数减一即可；数组 sz 只在根结点上有意义——`size_of(x)` 返回所在集合大小、`components()` 返回连通块个数。只想维护「集合大小」的话，上面这份模板已经够用，不需要额外结构。

合并时把大小相加、把连通块计数减一即可；数组 sz 只在根结点上有意义。

```c++
int size_of(int x) { return sz[find(x)]; }
int components() { return comps; }
```


### 应用


#### 带权并查集

数组 d 表示「点到父亲的权值」；路径压缩时一起累加，合并时按公式补上两个根之间的差。

```c++
int fa_w[N], d[N];                           // d[x]：x 到父亲的地址差
int find_w(int x) {
    if (fa_w[x] == x) return x;
    int r = find_w(fa_w[x]);
    d[x] += d[fa_w[x]];                      // 先递归再累加，保证是「到根的权值」
    return fa_w[x] = r;
}
// 合并「x 比 y 大 w」：rx = find(x), ry = find(y)
//   若 rx == ry：检查 d[x] - d[y] 是否等于 w
//   否则 fa_w[ry] = rx, d[ry] = d[x] - d[y] - w
bool merge_w(int x, int y, int w) {
    int rx = find_w(x), ry = find_w(y);
    if (rx == ry) return d[x] - d[y] == w;
    fa_w[ry] = rx, d[ry] = d[x] - d[y] - w;
    return true;
}
```


#### 扩展域并查集

把每个元素拆成 $k$ 个「域」，用「合并域」来表达「$x$ 属于哪一类」的约束；矛盾就是「两个不同的域落到了同一个集合」。

```c++
int n3;
int id(int x, int c) { return x + c * n3; }
void merge_all_diff(int x, int y) {          // 三种颜色，两两不同
    for (int c = 0; c < 3; c++)
        for (int t = 0; t < 3; t++)
            if (c != t) unite(id(x, c), id(y, t));
}
// 若某次合并时发现 x 的两个不同域已经在同一集合，说明出现矛盾
```


#### 可撤销并查集

要支持撤销就只能按秩合并、不做路径压缩（否则撤销无法还原）；用栈把「被改动的点与旧值」记下来。

```c++
int rfa[N], rsz[N], rk[N], htop;
pair<int, int> hist[N];                      // (被改动的点, 它的旧父亲)
int find_rev(int x) { return rfa[x] == x ? x : find_rev(rfa[x]); }   // 不压缩！
bool unite_rev(int a, int b) {
    int x = find_rev(a), y = find_rev(b);
    if (x == y) return false;
    if (rk[x] < rk[y]) swap(x, y);
    hist[htop++] = {y, rfa[y]};
    rfa[y] = x, rsz[x] += rsz[y];
    if (rk[x] == rk[y]) hist[htop++] = {x, rk[x]}, rk[x]++;   // 秩也要记录
    return true;
}
void undo_one() {
    auto [p, old] = hist[--htop];
    if (rfa[p] != p) { int f = rfa[p]; rsz[f] -= rsz[p]; }     // 撤销合并
    rfa[p] = old;
    // 若这条记录是「秩的变化」，就还原秩
}
// 实际实现里把「改动类型」也一起压栈，撤销时按类型分派即可
```


## 树状数组


### 一句话

给一个长度 $n\le 10^6$ 的序列和 $m$ 次操作，每次要么把 $a_i$ 加上 $x$、要么询问 $\sum_{i=l}^r a_i$（也可以把区间 $[l,r]$ 整体加 $x$ 后问某个位置的值，或求第 $k$ 小），修改与查询交替出现。**时间单次操作 $O(\log n)$，空间 $O(n)$。**


### 原理


### 实现

```c++
ll t[N];
int n;
void add(int i, ll v) { for (; i <= n; i += i & -i) t[i] += v; }
ll sum(int i) { ll r = 0; for (; i > 0; i -= i & -i) r += t[i]; return r; }
ll range_sum(int l, int r) { return sum(r) - sum(l - 1); }
```


### 应用


#### 区间修改与单点查询

把树状数组维护的数组设成差分数组：区间加变成两次单点修改，单点查就是前缀和。

```c++
void range_add(int l, int r, ll v) { add(l, v), add(r + 1, -v); }
ll point_get(int i) { return sum(i); }       // 差分的前缀和就是原值
```


#### 区间修改与区间查询

维护两个树状数组：一个存差分 $d_i$，一个存 $i\,d_i$；区间和可由这两个前缀和组合出来。

```c++
ll t1[N], t2[N];
void add2(int i, ll v) { for (int j = i; j <= n; j += j & -j) t1[j] += v, t2[j] += v * i; }
ll pre2(int i) { ll r = 0; for (int j = i; j > 0; j -= j & -j) r += t1[j] * (i + 1) - t2[j]; return r; }
void range_add2(int l, int r, ll v) { add2(l, v), add2(r + 1, -v); }
ll range_sum2(int l, int r) { return pre2(r) - pre2(l - 1); }
```


#### 二维树状数组

把一维的 add 与 sum 各套一层循环，复杂度 $O(\log^2 n)$，空间 $O(n^2)$（$n$ 大时要另想办法）。

```c++
ll t2d[N][N];
int n2d;
void add2d(int x, int y, ll v) {
    for (int i = x; i <= n2d; i += i & -i)
        for (int j = y; j <= n2d; j += j & -j) t2d[i][j] += v;
}
ll sum2d(int x, int y) {
    ll r = 0;
    for (int i = x; i > 0; i -= i & -i)
        for (int j = y; j > 0; j -= j & -j) r += t2d[i][j];
    return r;
}
ll rect_sum(int x1, int y1, int x2, int y2) {
    return sum2d(x2, y2) - sum2d(x1 - 1, y2) - sum2d(x2, y1 - 1) + sum2d(x1 - 1, y1 - 1);
}
```


#### 权值树状数组与树状数组上二分

把「值」当下标建树状数组，用倍增在 $O(\log n)$ 内找到第一个「前缀和不小于 $k$」的位置。

```c++
int kth(int k) {                             // 假设值域是 1..n
    int pos = 0;
    for (int pw = 1 << 20; pw; pw >>= 1)
        if (pos + pw <= n && t[pos + pw] < k) pos += pw, k -= t[pos];
    return pos + 1;
}
// 配合离散化就能求「动态第 k 小」
```


## 线段树


### 一句话

给一个长度 $n\le 10^6$ 的序列和 $m$ 次操作，每次要么把区间 $[l,r]$ 整体加 $x$ 或整体赋成 $x$，要么询问 $[l,r]$ 的和与最值、或在树上二分找第一个满足条件的位置，修改与查询交替出现。**时间单次操作 $O(\log n)$，空间 $O(n)$。**


### 原理


### 实现

```c++
ll seg[N << 2], tag[N << 2];
int n;
void pushup(int o) { seg[o] = seg[o << 1] + seg[o << 1 | 1]; }
void pushdown(int o, int l, int r) {
    if (!tag[o]) return;
    int mid = (l + r) >> 1;
    seg[o << 1] += tag[o] * (mid - l + 1), tag[o << 1] += tag[o];
    seg[o << 1 | 1] += tag[o] * (r - mid), tag[o << 1 | 1] += tag[o];
    tag[o] = 0;
}
void build(int o, int l, int r, ll* a) {
    if (l == r) { seg[o] = a[l]; return; }
    int mid = (l + r) >> 1;
    build(o << 1, l, mid, a), build(o << 1 | 1, mid + 1, r, a);
    pushup(o);
}
void update(int o, int l, int r, int ql, int qr, ll v) {
    if (ql <= l && r <= qr) { seg[o] += v * (r - l + 1), tag[o] += v; return; }
    pushdown(o, l, r);
    int mid = (l + r) >> 1;
    if (ql <= mid) update(o << 1, l, mid, ql, qr, v);
    if (qr > mid) update(o << 1 | 1, mid + 1, r, ql, qr, v);
    pushup(o);
}
ll query(int o, int l, int r, int ql, int qr) {
    if (ql <= l && r <= qr) return seg[o];
    pushdown(o, l, r);
    int mid = (l + r) >> 1;
    ll res = 0;
    if (ql <= mid) res += query(o << 1, l, mid, ql, qr);
    if (qr > mid) res += query(o << 1 | 1, mid + 1, r, ql, qr);
    return res;
}
```


### 应用


#### 标记永久化

修改时把标记留在结点上、不往下传；查询时把沿途的标记累加进答案。适合「可持久化」或「标记不好合并」的场景。

```c++
ll segp[N << 2], tg[N << 2];
void update_p(int o, int l, int r, int ql, int qr, ll v) {
    if (ql <= l && r <= qr) { tg[o] += v; return; }
    int mid = (l + r) >> 1;
    segp[o] += v * (min(r, qr) - max(l, ql) + 1);   // 当前结点覆盖到的部分直接加上
    if (ql <= mid) update_p(o << 1, l, mid, ql, qr, v);
    if (qr > mid) update_p(o << 1 | 1, mid + 1, r, ql, qr, v);
}
ll query_p(int o, int l, int r, int ql, int qr, ll acc) {
    if (ql <= l && r <= qr) return segp[o] + acc * (r - l + 1);
    acc += tg[o];                            // 沿途把标记带下去
    int mid = (l + r) >> 1;
    ll res = 0;
    if (ql <= mid) res += query_p(o << 1, l, mid, ql, qr, acc);
    if (qr > mid) res += query_p(o << 1 | 1, mid + 1, r, ql, qr, acc);
    return res;
}
```


#### 权值线段树与动态开点

把「值域」当下标建线段树；值域很大时只开用到的结点，单次操作 $O(\log V)$、空间 $O(m\log V)$。

```c++
int ls[N * 20], rs[N * 20], cntv[N * 20], tot, root;
void upd(int& o, int l, int r, int pos, int v) {
    if (!o) o = ++tot;
    cntv[o] += v;
    if (l == r) return;
    int mid = (l + r) >> 1;
    if (pos <= mid) upd(ls[o], l, mid, pos, v);
    else upd(rs[o], mid + 1, r, pos, v);
}
int kth(int o, int l, int r, int k) {
    if (l == r) return l;
    int mid = (l + r) >> 1;
    if (cntv[ls[o]] >= k) return kth(ls[o], l, mid, k);
    return kth(rs[o], mid + 1, r, k - cntv[ls[o]]);
}
```


#### zkw 线段树

把线段树按完全二叉树的下标建在数组里，自底向上做；区间操作时用左右两个指针往中间夹。

```c++
ll zseg[N << 1];
int zn;
void zbuild(int n_, ll* a) {
    zn = 1;
    while (zn < n_) zn <<= 1;
    for (int i = 1; i <= n_; i++) zseg[zn + i - 1] = a[i];
    for (int i = zn - 1; i >= 1; i--) zseg[i] = zseg[i << 1] + zseg[i << 1 | 1];
}
void zadd(int p, ll v) { for (int i = zn + p - 1; i; i >>= 1) zseg[i] += v; }
ll zsum(int l, int r) {
    ll res = 0;
    for (int i = zn + l - 1, j = zn + r + 1; i ^ j ^ 1; i >>= 1, j >>= 1) {
        if (~i & 1) res += zseg[i ^ 1];
        if (j & 1) res += zseg[j ^ 1];
    }
    return res;
}
```


#### 线段树二分

在线段树上「往左找、左边不够就往右走」，$O(\log n)$ 找到第一个满足条件的位置。

```c++
ll pre[N << 2];                              // 存区间和
int first_ge(int o, int l, int r, int ql, ll k) {
    // 找 [ql, n] 内第一个位置，使该位置之前（含）的和 >= k
    if (r < ql || pre[o] < k) return -1;     // 这一段完全用不上
    if (l == r) return l;
    int mid = (l + r) >> 1;
    int left = (ql <= mid) ? first_ge(o << 1, l, mid, ql, k) : -1;
    if (left != -1) return left;
    ll used = (ql <= mid) ? pre[o << 1] : 0; // 左边已经吃掉的量
    return first_ge(o << 1 | 1, mid + 1, r, ql, k - used);
}
```


## 线段树合并与分裂


### 一句话

给若干棵动态开点的权值线段树（总共 $O(n\log V)$ 个结点），要你把两棵树对应位置上的信息并到一起、或把一棵树按排名 $k$ 劈成两棵，合并与分裂交替出现。**时间合并总复杂度 $O(n\log V)$、单次分裂 $O(\log V)$，空间 $O(n\log V)$。**


### 原理


### 实现

```c++
int ls[N * 20], rs[N * 20], cntv[N * 20], tot;
int merge(int x, int y, int l, int r) {      // 合并：有一边为空就直接接上
    if (!x || !y) return x | y;
    if (l == r) { cntv[x] += cntv[y]; return x; }
    int mid = (l + r) >> 1;
    ls[x] = merge(ls[x], ls[y], l, mid);
    rs[x] = merge(rs[x], rs[y], mid + 1, r);
    cntv[x] = cntv[ls[x]] + cntv[rs[x]];
    return x;
}
// 分裂：把 x 里「值域最小的 k 个元素」留在 x，其余挪到新树 y
// 注意 x 必须传引用！分裂要改写 x 的左右儿子，按值传递改动会全部丢失
// x 保留值最小的 k 个，y 得到其余；调用后 y 是「另一棵树的根」
// 注意 x、y 都必须传引用：分裂要改写 x 的左右儿子
void split(int& x, int& y, int l, int r, int k) {
    y = 0;
    if (!x || k >= cntv[x]) return;          // x 全留，y 空
    if (k <= 0) { y = x, x = 0; return; }    // x 全给 y
    int mid = (l + r) >> 1;
    int lc = cntv[ls[x]];
    if (k < lc) {                            // 分裂点在左子树内部
        y = ++tot;
        ls[y] = rs[y] = cntv[y] = 0;
        split(ls[x], ls[y], l, mid, k);
        rs[y] = rs[x], rs[x] = 0;            // y 接管整个右子树
    } else if (k == lc) {                    // 正好卡在左右分界上，单独处理
        y = ++tot;                           // 仍然要给 y 开一个覆盖 [l,r] 的结点！
        ls[y] = rs[y] = cntv[y] = 0;
        rs[y] = rs[x], rs[x] = 0;            // x 留整棵左子树，y 的右子树接管过来
        // 此时 ls[y] 保持 0：y 在 [l,mid] 上没有任何元素
    } else {                                 // 分裂点在右子树内部
        y = ++tot;
        ls[y] = rs[y] = cntv[y] = 0;
        split(rs[x], rs[y], mid + 1, r, k - lc);
        // x 的左子树整体留下，ls[y] 保持 0
    }
    cntv[x] = cntv[ls[x]] + cntv[rs[x]];
    cntv[y] = cntv[ls[y]] + cntv[rs[y]];
}
```


**总的结点数怎么估**
每插入一个元素开 $O(\log V)$ 个结点；合并**不会新建**结点（合并时有一边为空就直接接上），只有分裂会新建 $O(\log V)$ 个。

```c++
// 结论：合并的总复杂度是 O(总结点数)，因为每次成功合并都会「消掉」一个结点
// 于是开数组时按 (插入次数 + 分裂次数) * logV 来估即可
const int LOGV = 17;                         // V = 1e5 时 ceil(log2 V)
int estimate_nodes(int insert_cnt, int split_cnt) {
    return (insert_cnt + split_cnt) * LOGV + 5;
}
// 例：n = m = 1e5 时约 3.4e6 个结点；每个结点 3 个 int，约 40 MB
// 注意「合并」本身不新增结点，所以不要把合并次数也算进去
```


**线段树合并**（并入自 13 章）
每个点开一棵动态开点权值线段树，自底向上合并。每次合并时有一边为空就整棵接上，所以总复杂度 $O(n\log V)$（每个结点最多被合并掉一次）。下面维护的是「权值出现次数」。

```c++
int ls[NN], rs[NN], val[NN], root[N], tot;
void upd(int& x, int l, int r, int pos, int v) {
    if (!x) x = ++tot;
    if (l == r) { val[x] += v; return; }
    int mid = (l + r) >> 1;
    if (pos <= mid) upd(ls[x], l, mid, pos, v);
    else upd(rs[x], mid + 1, r, pos, v);
    val[x] = val[ls[x]] + val[rs[x]];
}
int merge(int x, int y, int l, int r) {
    if (!x || !y) return x | y;               // 有一边为空直接接上
    if (l == r) { val[x] += val[y]; return x; }
    int mid = (l + r) >> 1;
    ls[x] = merge(ls[x], ls[y], l, mid);
    rs[x] = merge(rs[x], rs[y], mid + 1, r);
    val[x] = val[ls[x]] + val[rs[x]];
    return x;
}
int query(int x, int l, int r, int ql, int qr) {
    if (!x || qr < l || r < ql) return 0;
    if (ql <= l && r <= qr) return val[x];
    int mid = (l + r) >> 1;
    return query(ls[x], l, mid, ql, qr) + query(rs[x], mid + 1, r, ql, qr);
}
```

### 应用


#### 按值域分裂（序列分裂）
把线段树当「可分裂的平衡树」用：按值域 $[1,mid]$ 分成两棵，分别对应两段序列。

```c++
void split_value(int& x, int& y, int l, int r, int mid) {   // 同样要传引用
    if (!x) { y = 0; return; }
    if (r <= mid) { y = x, x = 0; return; } // 整棵都在左边
    if (l > mid) { y = 0; return; }         // 整棵都在右边，什么都不用挪
    y = ++tot;
    int m = (l + r) >> 1;
    split_value(ls[x], ls[y], l, m, mid);
    split_value(rs[x], rs[y], m + 1, r, mid);
    cntv[x] = cntv[ls[x]] + cntv[rs[x]];
    cntv[y] = cntv[ls[y]] + cntv[rs[y]];
}
```


#### 树上启发式合并统计子树信息
每个点一棵权值线段树，自底向上合并；合并后就能查任意子树的「第 $k$ 小 / 值域计数」。

```c++
int rt[N], counts[N * 20];
void merge_up(int u, int p) {                // 在树上自底向上合并
    for (int v : g[u]) {
        if (v == p) continue;
        merge_up(v, u);
        rt[u] = merge(rt[u], rt[v], 1, V);   // 直接把儿子的树并进来
    }
}
int subtree_kth(int u, int k) { return kth(rt[u], 1, V, k); }   // 子树第 k 小
int subtree_max(int o, int l, int r) {       // 子树最大权值：线段树上二分找最右非零
    if (cntv[o] == 0) return -1;
    if (l == r) return l;
    int mid = (l + r) >> 1;
    if (cntv[rs[o]]) return subtree_max(rs[o], mid + 1, r);
    return subtree_max(ls[o], l, mid);
}
```


## 线段树分治


### 一句话

给一张动态图（$n$ 个点 $m$ 条边，每条边只在时间区间 $[l,r]$ 内存在）和 $q\le 10^5$ 次询问，每条边在自己的生存期里被挂到时间线段树上，你带着并查集一路 DFS、进结点时加边、回溯时撤销，问 $a$ 与 $b$ 每个时刻是否连通 / 是否是二分图，加边与撤销交替出现。**时间 $O((m+q)\log q\log n)$，空间 $O(n+m)$。**


### 原理


### 实现

（把「每条边存在的时间区间」挂到时间线段树上，DFS 一遍时间线段树，进入结点时加边、离开时撤销。）

```c++
// 可撤销并查集（见「并查集」一节）：不能路径压缩
int rfa[N], rsz[N], rk[N], htop;
pair<int, int> hist[N * 40];
int find_rev(int x) { return rfa[x] == x ? x : find_rev(rfa[x]); }
bool unite_rev(int a, int b) {
    int x = find_rev(a), y = find_rev(b);
    if (x == y) return false;
    if (rk[x] < rk[y]) swap(x, y);
    hist[htop++] = {y, rfa[y]};
    rfa[y] = x, rsz[x] += rsz[y];
    if (rk[x] == rk[y]) hist[htop++] = {x, rk[x]}, rk[x]++;
    return true;
}
// 时间线段树：每个结点存「在这段时间内始终存在的边」
vector<pair<int, int>> te[N << 2];
void insert_edge(int o, int l, int r, int ql, int qr, pair<int, int> e) {
    if (ql <= l && r <= qr) { te[o].push_back(e); return; }
    int mid = (l + r) >> 1;
    if (ql <= mid) insert_edge(o << 1, l, mid, ql, qr, e);
    if (qr > mid) insert_edge(o << 1 | 1, mid + 1, r, ql, qr, e);
}
int ans[N];
void dfs_time(int o, int l, int r) {
    int snap = htop;
    for (auto [u, v] : te[o]) unite_rev(u, v);
    if (l == r) {
        ans[l] = 0;                          // 当前并查集给出的答案，例如连通块个数
    } else {
        int mid = (l + r) >> 1;
        dfs_time(o << 1, l, mid);
        dfs_time(o << 1 | 1, mid + 1, r);
    }
    while (htop > snap) {                    // 撤销到进入时的状态
        auto [p, old] = hist[--htop];
        if (rfa[p] != p) rsz[rfa[p]] -= rsz[p];
        rfa[p] = old;
    }
}
```


### 应用


#### 动态图连通性（离线）
每条边给出它的存在区间，按上面的流程跑一遍，每个时刻的连通块个数就是答案。

```c++
int comps = n;
bool unite_cnt(int a, int b) {
    int x = find_rev(a), y = find_rev(b);
    if (x == y) return false;                // 没合并，不用记
    if (rk[x] < rk[y]) swap(x, y);
    hist[htop++] = {y, rfa[y]};
    rfa[y] = x, rsz[x] += rsz[y], comps--;   // 连通块数减一
    if (rk[x] == rk[y]) hist[htop++] = {x, rk[x]}, rk[x]++;
    return true;
}
// 撤销时必须把 comps 加回来：只要 hist 里这一条确实是「成功的合并」
// 所以记录里要带上「类型」标记，或者用 comps 与 htop 的关系还原
```


#### 二分图判定（离线）
用「扩展域并查集」：把每个点拆成「左侧 / 右侧」两个域，一条边就把 u 的左与 v 的右、u 的右与 v 的左合并；若某次合并发现同侧，说明出现奇环、不是二分图。

```c++
// 域编号：id(u,0) = u, id(u,1) = u + n
bool unite_bip(int u, int v) {
    if (find_rev(u) == find_rev(v)) return false;   // 已经在同一侧 -> 出现奇环
    unite_rev(u, v + n), unite_rev(u + n, v);
    return true;
}
// 同样把「当前是不是二分图」做成可回滚的全局标记
```


## 李超树


### 一句话

一开始有一堆直线 $y=kx+b$，要你不断插入新的直线（或只在 $[l,r]$ 内有效的线段），并询问某个横坐标 $x$ 处所有直线的最大值 / 最小值，插入与查询交替出现，其中 $C$ 是横坐标范围。**时间单次操作 $O(\log^2 C)$（线段插入）或 $O(\log C)$（整条直线），空间 $O(n\log C)$。**


### 原理


### 实现

（在值域线段树的每个结点上存一条「可能最优」的直线；插入时逐层比较、把较劣的那条留在结点上。）

```c++
// 李超树：求区间最大值。要最小值就把 k、b 取负，或者把比较号反过来
struct Line { ll k, b; };
ll f(Line l, ll x) { return l.k * x + l.b; }
Line tr[N << 2];
bool has[N << 2];

void insert_line(int o, int l, int r, Line nw) {
    if (!has[o]) { tr[o] = nw, has[o] = true; return; }
    int mid = (l + r) >> 1;
    if (f(nw, mid) > f(tr[o], mid)) swap(tr[o], nw);   // 让 tr[o] 在中点更优
    if (l == r) return;
    // nw 在中点不占优，但它可能在某一端占优
    if (f(nw, l) > f(tr[o], l)) insert_line(o << 1, l, mid, nw);
    else if (f(nw, r) > f(tr[o], r)) insert_line(o << 1 | 1, mid + 1, r, nw);
}
ll query(int o, int l, int r, int x) {
    ll res = has[o] ? f(tr[o], x) : -(ll)4e18;
    if (l == r) return res;
    int mid = (l + r) >> 1;
    if (x <= mid) return max(res, query(o << 1, l, mid, x));
    return max(res, query(o << 1 | 1, mid + 1, r, x));
}
```


### 应用


#### 求最小值
把每条直线取负后照常插入（查询时再取负），或者在比较函数里把符号全部反过来。

```c++
void insert_min(Line nw) {                   // 求最小值：取负后当最大值插入
    insert_line(1, XMIN, XMAX, {-nw.k, -nw.b});
}
ll query_min(int x) { return -query(1, XMIN, XMAX, x); }
// 溢出：k 到 1e9、x 到 1e6 时乘积到 1e15，long long 够用；
// 但若 k 到 1e18，就要把 f() 换成 __int128
```


#### 线段插入（只在 $[l,r]$ 内有效）
把「线段」拆到 $O(\log C)$ 个结点上，每个结点插入整条直线即可。

```c++

    if (ql <= l && r <= qr) { insert_line(o, l, r, nw); return; }
    int mid = (l + r) >> 1;

// 复杂度 O(log^2 C)
```


#### 斜率优化 DP
DP 转移形如 $dp_i=\min_j(k_j x_i+b_j)$ 且 $x_i$ 单调时，不用单调队列，直接把每个 $j$ 对应的直线插进李超树。

```c++
ll dp[N], pre[N];
void slope_opt(int n) {
    for (int i = 1; i <= n; i++) pre[i] = pre[i - 1] + a[i];
    memset(has, 0, sizeof has);
    insert_min({0, 0});                      // j = 0 时的直线：斜率 -2*pre[0] = 0，截距 0
    for (int i = 1; i <= n; i++) {
        dp[i] = pre[i] * pre[i] + query_min((int)pre[i]);
        insert_min({-2 * pre[i], dp[i] + pre[i] * pre[i]});
    }
}
// 好处：x 不需要单调，也不用处理「相等 / 负数」的边界
```


## Segment Tree Beats


### 一句话

给一个序列和 $m$ 次操作（$n,m\le 10^5$），每次要么把区间 $[l,r]$ 对 $x$ 取 $\min$ / 取 $\max$，要么询问 $[l,r]$ 的和与最值，区间加还要和取 min 的标记按固定顺序一起下传，修改与查询交替出现。**时间均摊 $O(n\log n)$ 内处理完 $m$ 次操作，空间 $O(n)$。**


### 原理


### 实现

（核心是「如果这个结点的最大值已经被取 min 取到不用再往下，就整体处理；否则递归」。）

```c++
ll sum[N << 2], mx[N << 2], se[N << 2], mxc[N << 2];   // se：次大值，mxc：最大值的个数
int n;

void pushup(int o) {
    sum[o] = sum[o << 1] + sum[o << 1 | 1];
    if (mx[o << 1] == mx[o << 1 | 1]) {
        mx[o] = mx[o << 1];
        se[o] = max(se[o << 1], se[o << 1 | 1]);
        mxc[o] = mxc[o << 1] + mxc[o << 1 | 1];
    } else if (mx[o << 1] > mx[o << 1 | 1]) {
        mx[o] = mx[o << 1];
        se[o] = max(se[o << 1], mx[o << 1 | 1]);
        mxc[o] = mxc[o << 1];
    } else {
        mx[o] = mx[o << 1 | 1];
        se[o] = max(se[o << 1 | 1], mx[o << 1]);
        mxc[o] = mxc[o << 1 | 1];
    }
}
// 把结点 o 的「最大值」压到 v（要求 v > se[o]）
void push_min(int o, ll v) { sum[o] -= (mx[o] - v) * mxc[o]; mx[o] = v; }
void pushdown(int o) {
    // 用父亲的 mx 去压儿子：儿子的大值若超过父亲的 mx，说明它应当被取 min
    if (mx[o << 1] > mx[o]) push_min(o << 1, mx[o]);
    if (mx[o << 1 | 1] > mx[o]) push_min(o << 1 | 1, mx[o]);
}
void range_min(int o, int l, int r, int ql, int qr, ll v) {
    if (qr < l || r < ql || mx[o] <= v) return;
    if (ql <= l && r <= qr && se[o] < v) { push_min(o, v); return; }   // 只影响最大值
    pushdown(o);
    int mid = (l + r) >> 1;
    range_min(o << 1, l, mid, ql, qr, v);
    range_min(o << 1 | 1, mid + 1, r, ql, qr, v);
    pushup(o);
}
```


### 应用


#### 区间取 max
对称地再维护一份「最小值与次小值」，就能处理区间取 max。

```c++
ll mn[N << 2], se2[N << 2], mnc[N << 2];     // 次小值、最小值的个数
void pushup2(int o) {                        // 与 pushup 对称，维护 mn / se2 / mnc
    if (mn[o << 1] == mn[o << 1 | 1]) {
        mn[o] = mn[o << 1];
        se2[o] = min(se2[o << 1], se2[o << 1 | 1]);
        mnc[o] = mnc[o << 1] + mnc[o << 1 | 1];
    } else if (mn[o << 1] < mn[o << 1 | 1]) {
        mn[o] = mn[o << 1], se2[o] = min(se2[o << 1], mn[o << 1 | 1]), mnc[o] = mnc[o << 1];
    } else {
        mn[o] = mn[o << 1 | 1], se2[o] = min(se2[o << 1 | 1], mn[o << 1]), mnc[o] = mnc[o << 1 | 1];
    }
}
void push_max(int o, ll v) { sum[o] += (v - mn[o]) * mnc[o]; mn[o] = v; }
void range_max(int o, int l, int r, int ql, int qr, ll v) {
    if (qr < l || r < ql || mn[o] >= v) return;
    if (ql <= l && r <= qr && se2[o] > v) { push_max(o, v); return; }
    pushdown(o);
    int mid = (l + r) >> 1;
    range_max(o << 1, l, mid, ql, qr, v);
    range_max(o << 1 | 1, mid + 1, r, ql, qr, v);
    pushup(o), pushup2(o);
}
```


#### 区间加 + 区间取 min（务必对拍）
区间加要乘的是**区间长度**（作用到所有元素），不是最大值的个数——这一点极易写错：

```c++
ll len[N << 2], addtag[N << 2];              // len 在 build 里填成 r - l + 1
// 区间加：所有元素一起加，所以乘 len
void push_add(int o, ll v) { sum[o] += v * len[o]; mx[o] += v; se[o] += v; addtag[o] += v; }
// 区间取 min：只有「等于最大值的那些元素」被压低，所以乘 mxc
void push_min_tag(int o, ll v) { sum[o] -= (mx[o] - v) * mxc[o]; mx[o] = v; }
void pushdown_with_add(int o) {
    if (addtag[o]) {
        push_add(o << 1, addtag[o]), push_add(o << 1 | 1, addtag[o]);
        addtag[o] = 0;
    }
    if (mx[o << 1] > mx[o]) push_min_tag(o << 1, mx[o]);
    if (mx[o << 1 | 1] > mx[o]) push_min_tag(o << 1 | 1, mx[o]);
}
```

这两条标记的**下传顺序与相互影响**比看上去微妙：区间加会同时改动 mx 与 se，而取 min 只压最大值，一旦顺序或乘的量写错，错误会在多次操作后以「数值整体偏移」的形式暴露出来。**这一版请务必和暴力对拍**（上面的 pushdown 已验证在「只有取 min」时正确；加入加法标记后要重新对拍）。

```c++
// 对拍模板：随机 n <= 30，随机做 区间加 / 区间取 min / 区间求和，
// 每次把线段树每个叶子的值与暴力数组比较（不要只比较总和！）
// 只比较总和时，两处相反的错误可能互相抵消而看不出来
```


#### 区间最值与历史最值
除了当前值，再维护「历史最大值」；下传时用「加法标记的历史最大值」去更新儿子的历史最大。

```c++

// 每次 push_add 时：
//   hmx[o] = max(hmx[o], mx[o] + hadd[o])   // 用「加之前的最大附加量」更新历史
```

## 猫树


### 一句话

给一个不会改动的静态序列（$n\le 10^5$，运算可结合，如 max、gcd、矩阵乘）和 $q\le 10^5$ 次区间查询，每次问 $[l,r]$ 的区间信息，预处理时把以中点为界的两侧前缀 / 后缀结果都存下来，预处理与查询交替出现。**时间预处理 $O(n\log n)$、单次查询 $O(1)$，空间 $O(n\log n)$。**


### 原理


### 实现

（猫树：在每个分治区间里预处理「中点到两端的合并结果」，查询时取 $l$、$r$ 的 LCA 所在层直接合并。）

```c++
const int LG = 18;                           // 2^LG >= n，必须补到 2 的幂！
const int SZ = 1 << (LG + 1);
ll cat[LG][SZ];                              // cat[dep][i]：第 dep 层里，i 到本区间中点的合并值
int pos[SZ];                                 // pos[i]：i 对应的叶子在隐式堆里的编号
int a[SZ], n;

void build(int o, int l, int r, int dep) {
    if (l == r) { pos[l] = o; return; }
    int mid = (l + r) >> 1;
    cat[dep][mid] = a[mid];
    for (int i = mid - 1; i >= l; i--) cat[dep][i] = max(cat[dep][i + 1], (ll)a[i]);
    cat[dep][mid + 1] = a[mid + 1];          // 补到 2 的幂后，mid+1 一定 <= r
    for (int i = mid + 2; i <= r; i++) cat[dep][i] = max(cat[dep][i - 1], (ll)a[i]);
    build(o << 1, l, mid, dep + 1);
    build(o << 1 | 1, mid + 1, r, dep + 1);
}
ll cat_query(int l, int r) {
    if (l == r) return a[l];
    // pos[l] ^ pos[r] 的最高位给出 LCA：但它是「从最深层往上数」的层号
    int k = 31 - __builtin_clz(pos[l] ^ pos[r]);
    int dep = LG - 1 - k;                    // 换算成「从根往下数」的深度
    return max(cat[dep][l], cat[dep][r]);
}
// 初始化：把 n+1..(1<<LG) 都填成 -inf，再 build(1, 1, 1 << LG, 0)
```


### 应用


#### 区间 max


#### 区间 gcd
把 cat 数组的合并运算换成 gcd、min 或矩阵乘法即可；要求运算是**可结合**的（不要求可减）。

```c++
ll catg[LOG][N];
void build_gcd(int o, int l, int r, int dep) {
    if (l == r) { pos[l] = o; return; }
    int mid = (l + r) >> 1;
    catg[dep][mid] = a[mid];
    for (int i = mid - 1; i >= l; i--) catg[dep][i] = std::gcd(catg[dep][i + 1], (ll)a[i]);
    if (mid + 1 <= r) catg[dep][mid + 1] = a[mid + 1];
    for (int i = mid + 2; i <= r; i++) catg[dep][i] = std::gcd(catg[dep][i - 1], (ll)a[i]);
    build_gcd(o << 1, l, mid, dep + 1);
    build_gcd(o << 1 | 1, mid + 1, r, dep + 1);
}
ll gcd_query(int l, int r) {
    if (l == r) return a[l];
    int k = 31 - __builtin_clz(pos[l] ^ pos[r]);
    return std::gcd(catg[k][l], catg[k][r]);
}
// max、gcd、min、矩阵乘都可以；但「求和」不行——两段合起来会重复 / 缺项
```


## 平衡树


### 一句话

维护一个有序多重集（$n\le 10^5$，值域 $\le 10^9$），要不断插入 $x$、删除 $x$、问 $x$ 的排名、问第 $k$ 小是谁、问 $x$ 的前驱后继，或把一段区间整个翻转，插入与查询交替出现。**时间单次操作均摊 $O(\log n)$，空间 $O(n)$。**


### 原理


### 实现

（Fhq Treap：用随机优先级 + 分裂 / 合并实现所有操作，不需要旋转，天然支持可持久化与区间操作。）

```c++
int ls[N], rs[N], val[N], pr[N], sz[N], tot;
mt19937 rng(chrono::steady_clock::now().time_since_epoch().count());

int newnode(int v) { val[++tot] = v, pr[tot] = rng(), sz[tot] = 1, ls[tot] = rs[tot] = 0; return tot; }
void pushup(int o) { if (o) sz[o] = sz[ls[o]] + sz[rs[o]] + 1; }
// 按值分裂：左树所有值 <= v，右树所有值 > v
void split(int o, int v, int& x, int& y) {
    if (!o) { x = y = 0; return; }
    if (val[o] <= v) { x = o, split(rs[o], v, rs[o], y); }
    else { y = o, split(ls[o], v, x, ls[o]); }
    pushup(o);
}
// 按大小分裂：左树恰好 k 个
void split_k(int o, int k, int& x, int& y) {
    if (!o) { x = y = 0; return; }
    if (sz[ls[o]] + 1 <= k) { x = o, split_k(rs[o], k - sz[ls[o]] - 1, rs[o], y); }
    else { y = o, split_k(ls[o], k, x, ls[o]); }
    pushup(o);
}
int merge(int x, int y) {
    if (!x || !y) return x | y;
    if (pr[x] < pr[y]) { rs[x] = merge(rs[x], y); pushup(x); return x; }
    ls[y] = merge(x, ls[y]); pushup(y); return y;
}
```


### 应用


#### 区间翻转（文艺平衡树）

把序列按位置放进平衡树，翻转区间就相当于「分裂出这段、打上翻转标记、再合回去」。

```c++
bool rev[N];
void pushdown(int o) {
    if (!rev[o]) return;
    swap(ls[o], rs[o]);
    if (ls[o]) rev[ls[o]] ^= 1;
    if (rs[o]) rev[rs[o]] ^= 1;
    rev[o] = false;
}
void reverse_range(int& root, int l, int r) {
    int a, b, c;
    split_k(root, r, a, c);
    split_k(a, l - 1, a, b);
    rev[b] ^= 1;                             // 整段打标记
    root = merge(merge(a, b), c);
}
// 中序遍历输出即为当前序列；每个操作前后记得 pushdown
```


#### Treap


三种经典平衡树各有侧重：Treap 靠随机优先级旋转、Splay 靠「伸展」维护均摊复杂度、替罪羊树靠「重构不平衡子树」。

```c++
// ---------- 旋转 Treap：插入 + 旋转 ----------
int rls[N], rrs[N], rval[N], rpr[N], rsz[N], rtot;
void rot(int& o, int d) {                    // d=0 左旋，d=1 右旋
    int t = d ? rls[o] : rrs[o];
    if (d) rls[o] = rrs[t], rrs[t] = o;
    else rrs[o] = rls[t], rls[t] = o;
    rsz[t] = rsz[o];
    rsz[o] = rsz[rls[o]] + rsz[rrs[o]] + 1;
    o = t;
}
void insert_rot(int& o, int v) {
    if (!o) { o = ++rtot, rval[o] = v, rpr[o] = rng() % 100000, rsz[o] = 1; return; }
    if (v < rval[o]) {
        insert_rot(rls[o], v);
        if (rpr[rls[o]] < rpr[o]) rot(o, 1);
    } else {
        insert_rot(rrs[o], v);
        if (rpr[rrs[o]] < rpr[o]) rot(o, 0);
    }
    rsz[o] = rsz[rls[o]] + rsz[rrs[o]] + 1;
}
```


#### 无旋 Treap（FHQ Treap）

本模块「实现」一节给出的就是无旋 Treap（FHQ Treap）：用随机优先级 + 分裂 / 合并完成全部操作，不需要旋转，因此天然支持可持久化与区间操作。


#### Splay

（把访问到的点一层层旋到根，靠**均摊**保证单次 $O(\log n)$。）

```c++
int sp_ch[N][2], sp_fa[N], sp_val[N], sp_sz[N], sp_cnt[N], sp_root, sp_tot;

void sp_pushup(int x) { sp_sz[x] = sp_sz[sp_ch[x][0]] + sp_sz[sp_ch[x][1]] + sp_cnt[x]; }

void sp_rotate(int x) {
    int y = sp_fa[x], z = sp_fa[y], k = (sp_ch[y][1] == x), w = sp_ch[x][k ^ 1];
    sp_ch[y][k] = w; if (w) sp_fa[w] = y;
    sp_ch[x][k ^ 1] = y, sp_fa[y] = x, sp_fa[x] = z;
    if (z) sp_ch[z][sp_ch[z][1] == y] = x;
    sp_pushup(y), sp_pushup(x);
}
void sp_splay(int x) {                        // 把 x 旋到根
    while (sp_fa[x]) {
        int y = sp_fa[x], z = sp_fa[y];
        // 一字型（x、y、z 同一方向）要「先旋 y 再旋 x」，否则均摊代价不成立
        if (z) sp_rotate(((sp_ch[y][0] == x) ^ (sp_ch[z][0] == y)) ? x : y);
        sp_rotate(x);
    }
    sp_root = x;
}
// 因为能把任意点旋到根，Splay 特别适合「提取区间」：
//   把 l-1 旋到根、r+1 旋到根的右儿子，根的右儿子的左子树就是 [l, r]
void sp_insert(int v) {
    int x = sp_root, p = 0;
    while (x && sp_val[x] != v) p = x, x = sp_ch[x][sp_val[x] < v];
    // 注意：插入重复值时只加 cnt，但 sp_sz 也要更新；
    // 若 x 本来就是根，splay 的循环一次都不执行，必须显式 pushup 一次
    if (x) { sp_cnt[x]++; sp_splay(x); sp_pushup(x); return; }
    x = ++sp_tot, sp_val[x] = v, sp_sz[x] = sp_cnt[x] = 1, sp_fa[x] = p;
    if (p) sp_ch[p][sp_val[p] < v] = x;
    sp_splay(x);
}
```

#### 替罪羊树

（不旋转。插入后沿路检查「子树是否太不平衡」，一旦超过阈值 $\alpha$ 就把该子树中序拍平后重建；重建是均摊 $O(1)$ 的。）

```c++
const double ALPHA = 0.75;
int go_ls[N], go_rs[N], go_val[N], go_sz[N], go_tot, go_root;
vector<int> flat;

void go_pushup(int x) { go_sz[x] = go_sz[go_ls[x]] + go_sz[go_rs[x]] + 1; }
bool too_unbalanced(int x) {                  // 一边的占比超过了 ALPHA
    return go_sz[go_ls[x]] > ALPHA * go_sz[x] || go_sz[go_rs[x]] > ALPHA * go_sz[x];
}
void flatten(int x) {                         // 中序遍历，把结点收进 flat
    if (!x) return;
    flatten(go_ls[x]);
    flat.push_back(x);
    flatten(go_rs[x]);
}
int rebuild(int l, int r) {                   // 把 flat[l..r] 重建成完全平衡
    if (l > r) return 0;
    int mid = (l + r) >> 1, x = flat[mid];
    go_ls[x] = rebuild(l, mid - 1);
    go_rs[x] = rebuild(mid + 1, r);
    go_pushup(x);
    return x;
}
// x 传引用：重建后改的正好是父结点里那个儿子指针，不用额外「挂回去」
void go_insert(int& x, int v) {
    if (!x) { x = ++go_tot, go_val[x] = v, go_sz[x] = 1; return; }
    go_insert(v < go_val[x] ? go_ls[x] : go_rs[x], v);
    go_pushup(x);
    if (too_unbalanced(x)) {
        flat.clear();
        flatten(x);
        x = rebuild(0, (int)flat.size() - 1);
    }
}
```

#### 可持久化平衡树

Fhq Treap 的分裂 / 合并是「自顶向下、只改一条链」，把改动过的结点都复制一份就是可持久化。

```c++
int copy_node(int o) {
    if (!o) return 0;
    ++tot;
    ls[tot] = ls[o], rs[tot] = rs[o], val[tot] = val[o], pr[tot] = pr[o], sz[tot] = sz[o];
    return tot;
}
void split_p(int o, int v, int& x, int& y) {
    if (!o) { x = y = 0; return; }
    int nw = copy_node(o);
    if (val[o] <= v) { x = nw, split_p(rs[nw], v, rs[nw], y); }
    else { y = nw, split_p(ls[nw], v, x, ls[nw]); }
    pushup(nw);
}
// 每次操作只新建 O(log n) 个结点，空间 O((n+m)log n)
```


## 可持久化数据结构


### 一句话

给一个序列和 $m$ 次操作（$n,m\le 10^5$），每次要么基于历史版本把某个位置改成 $x$ 得到新版本、之后还能回头查询旧版本，要么静态地问区间 $[l,r]$ 内与 $x$ 异或的最大值，修改与查询交替出现。**时间单次操作 $O(\log n)$，空间总共 $O((n+m)\log n)$。**


### 原理


### 实现

只复制「被改动的那条链」，其余结点共用 —— 这就是全部思想。


### 应用


#### 可持久化线段树（主席树）
（核心：只复制「被改动的那条链」，其余结点共用。）

```c++
// ---------- 可持久化数组（可持久化线段树） ----------
int lc[N * 20], rc[N * 20], sumv[N * 20], cnt, root[N];
void build(int& o, int l, int r) {
    o = ++cnt;
    if (l == r) return;
    int mid = (l + r) >> 1;
    build(lc[o], l, mid), build(rc[o], mid + 1, r);
}
void update(int pre, int& o, int l, int r, int pos, int v) {
    o = ++cnt;
    lc[o] = lc[pre], rc[o] = rc[pre], sumv[o] = sumv[pre] + 1;   // 复制 + 改动
    if (l == r) return;
    int mid = (l + r) >> 1;
    if (pos <= mid) update(lc[pre], lc[o], l, mid, pos, v);
    else update(rc[pre], rc[o], mid + 1, r, pos, v);
}
```


#### 可持久化数组
用可持久化线段树维护数组，每次修改新建 $O(\log n)$ 个结点，就能「回到任意历史版本」。

```c++
// 建出初始版本；第 i 次修改得到版本 i
// 查询版本 v 的 a[pos]：
int query_arr(int o, int l, int r, int pos) {
    if (l == r) return sumv[o];              // 叶子直接存值
    int mid = (l + r) >> 1;
    if (pos <= mid) return query_arr(lc[o], l, mid, pos);
    return query_arr(rc[o], mid + 1, r, pos);
}
void modify(int pre, int& o, int l, int r, int pos, int v) {
    o = ++cnt;
    lc[o] = lc[pre], rc[o] = rc[pre];
    if (l == r) { sumv[o] = v; return; }
    int mid = (l + r) >> 1;
    if (pos <= mid) modify(lc[pre], lc[o], l, mid, pos, v);
    else modify(rc[pre], rc[o], mid + 1, r, pos, v);
}
// build(root[0]);  第 i 次修改：modify(root[i-1], root[i], 1, n, pos, v);
// 查询历史版本：query_arr(root[v], 1, n, pos)
```


#### 可持久化 Trie
对每个前缀建一棵 Trie，查询 $[l,r]$ 时用「版本 r 的 Trie 减去版本 l-1 的 Trie」判断某条分支是否存在。

```c++
int trie[N * 32][2], trie_cnt[N * 32], ttot, troot[N];
void tinsert(int pre, int& o, int x, int bit) {
    o = ++ttot;
    trie[o][0] = trie[pre][0], trie[o][1] = trie[pre][1];
    trie_cnt[o] = trie_cnt[pre] + 1;
    if (bit < 0) return;
    int b = (x >> bit) & 1;
    tinsert(trie[pre][b], trie[o][b], x, bit - 1);
}
int tquery_max(int lo, int ro, int x, int bit, int cur) {
    if (bit < 0) return cur;
    int b = ((x >> bit) & 1) ^ 1;            // 尽量走相反位
    if (trie_cnt[trie[ro][b]] - trie_cnt[trie[lo][b]] > 0)
        return tquery_max(trie[lo][b], trie[ro][b], x, bit - 1, cur | (1 << bit));
    return tquery_max(trie[lo][b ^ 1], trie[ro][b ^ 1], x, bit - 1, cur);
}
```


## 数列分块


### 一句话

给一个长度 $n\le 10^5$ 的序列和 $m\le 10^5$ 次操作，每次要么把区间 $[l,r]$ 整体加上 $x$、要么询问 $\sum_{i=l}^r a_i$，整块一起处理、边角逐个改，修改与查询交替出现。**时间单次操作 $O(\sqrt n)$，空间 $O(n)$。**


### 原理


### 实现

```c++
int n, bs, bn;                               // bs：块长，bn：块数
ll a[N], addt[N], sumb[N];
int bid[N];

void build(int n_) {
    n = n_;
    bs = max(1, (int)sqrt(n));
    bn = (n + bs - 1) / bs;
    for (int i = 1; i <= n; i++) bid[i] = (i - 1) / bs + 1;
    for (int i = 1; i <= n; i++) sumb[bid[i]] += a[i];
}
void range_add(int l, int r, ll v) {
    for (int i = l; i <= min(r, bid[l] * bs); i++) a[i] += v, sumb[bid[i]] += v;
    if (bid[l] != bid[r])
        for (int i = (bid[r] - 1) * bs + 1; i <= r; i++) a[i] += v, sumb[bid[i]] += v;
    for (int b = bid[l] + 1; b <= bid[r] - 1; b++) addt[b] += v, sumb[b] += v * bs;
}
ll range_sum(int l, int r) {
    ll res = 0;
    for (int i = l; i <= min(r, bid[l] * bs); i++) res += a[i] + addt[bid[i]];
    if (bid[l] != bid[r])
        for (int i = (bid[r] - 1) * bs + 1; i <= r; i++) res += a[i] + addt[bid[i]];
    for (int b = bid[l] + 1; b <= bid[r] - 1; b++) res += sumb[b];
    return res;
}
```


### 应用


#### 值域分块
把值域分成 $\sqrt V$ 块，先扫块找「第 $k$ 小落在哪个块」，再在块内逐个找，$O(\sqrt V)$ 一次查询。

```c++
int vcnt[N], vblk[N], vbs;                   // 值域 [1, V]
void vadd(int x, int v) { vcnt[x] += v, vblk[(x - 1) / vbs + 1] += v; }
int vkth(int k) {
    int b = 1;
    while (vblk[b] < k) k -= vblk[b], b++;   // 先定位到块
    for (int x = (b - 1) * vbs + 1; ; x++)
        if ((k -= vcnt[x]) <= 0) return x;   // 再在块内逐个减
}
// 复杂度 O(sqrt(V))，比树状数组的 O(log V) 慢，但常数小、好写、可撤销
```


#### 根号重构

（把「修改」先攒着，攒够 $B$ 个就整体重构一次；查询时「查已重构的部分 + 暴力扫攒下来的部分」。）

```c++
// 例：不断插入一个数，并随时询问「有多少个数小于 x」
const int B = 700;
vector<int> sorted;                           // 已经并进有序数组的部分
vector<int> pending;                          // 还没并进去的（最多 B 个）

void ins(int x) {
    pending.push_back(x);
    if ((int)pending.size() >= B) {           // 攒够了 -> 一次性重构
        sorted.insert(sorted.end(), pending.begin(), pending.end());
        sort(sorted.begin(), sorted.end());
        pending.clear();
    }
}
int count_less(int x) {
    int r = (int)(lower_bound(sorted.begin(), sorted.end(), x) - sorted.begin());
    for (int v : pending) if (v < x) r++;     // 攒下来的部分暴力扫，O(B)
    return r;
}
// 复杂度：m 次插入里重构 m/B 次，每次 O(n log n)；每次询问 O(B) 次比较。
// 取 B = sqrt(n log n) 即可让总复杂度落在 O(m sqrt(n log n))。
// 同一招还常用于「动态图连通性（只加边）」「带修改的区间查询」等。
```

#### 根号重构
维护「插入数 + 查询第 $k$ 小」：不每次插入都重建，而是攒够 $B$ 个新元素就整体重建一次；查询时把「有序旧数组的二分」与「缓冲区暴力扫」合起来。总复杂度 $O(n\sqrt n)$。

```c++
vector<int> sorted_, buffer;
int B2 = 700;
void rebuild_sqrt() {
    sorted_.insert(sorted_.end(), buffer.begin(), buffer.end());
    sort(sorted_.begin(), sorted_.end());
    buffer.clear();
}
void insert_sqrt(int x) {
    buffer.push_back(x);
    if ((int)buffer.size() >= B2) rebuild_sqrt();
}
int kth_sqrt(int k) {                        // 二分答案，数「旧数组 + 缓冲区」里 <= mid 的个数
    int lo = -1000000000, hi = 1000000000;
    while (lo < hi) {
        int mid = lo + (hi - lo) / 2;
        int c = upper_bound(sorted_.begin(), sorted_.end(), mid) - sorted_.begin();
        for (int x : buffer) c += (x <= mid);
        if (c >= k) hi = mid; else lo = mid + 1;
    }
    return lo;
}
```

## 根号分治


### 一句话

给 $n,q\le 10^5$ 个询问，拿阈值 $\sqrt n$ 把对象分成两类：相关的量超过 $\sqrt n$ 的按「重」的集中预处理、没超过的按「轻」的逐个暴力，预处理与暴力交替出现。**时间总共 $O(n\sqrt n)$，空间 $O(n)$。**


### 原理


### 实现

按「某个量是否超过 $\sqrt n$」分类：小的暴力枚举、大的预处理，两边合计 $O(n\sqrt n)$。

```c++
// 例：多次询问「有多少对 (i,j)（i<j）使 a_i + a_j = x」
// 记 cnt[v] = 值 v 的出现次数。分两类：
//   出现次数 > sqrt(n) 的「重数」不超过 sqrt(n) 个，预处理它们两两的贡献；
//   其余是「轻数」，枚举 x 的所有拆法 v + (x-v)，两边的出现次数相乘。
vector<int> heavy;
int cntv[N];
ll solve_heavy(int x) {
    ll r = 0;
    for (int v : heavy) {
        int w = x - v;
        if (w < 0 || w >= N) continue;
        if (v == w) r += (ll)cntv[v] * (cntv[v] - 1) / 2;
        else if (v < w) r += (ll)cntv[v] * cntv[w];
    }
    return r;
}
// 轻数的部分：只枚举「出现过的值」里小于 sqrt(n) 的那些，配对即可
```


**根号分治**（并入自 13 章）
按度数把点分成「大点」（度数大于 $B$，只有 $O(m/B)$ 个）与「小点」：给每个大点预处理一份「邻点颜色计数」，小点询问时直接枚举邻居。取 $B=\sqrt m$，两端都是 $O(\sqrt m)$，总复杂度 $O((n+q)\sqrt m)$。

```c++
    bigColor.assign(bigCnt, vector<int>(K + 1, 0));

        for (int v : g2[u]) bigColor[bigId[u]][color_[v]]++;
}

    if (big[u]) return bigColor[bigId[u]][c];
    int r = 0;
    for (int v : g2[u]) r += (color_[v] == c);
    return r;
}
```

## 珂朵莉树（ODT）


### 一句话

维护一个序列的「颜色段」集合（$n,m\le 10^5$），每次把区间 $[l,r]$ 整体赋成同一个值、或统计区间里的信息，赋值时把覆盖到的段并成一段，赋值与统计交替出现。**时间随机数据下均摊 $O(n\log\log n)$ 一次赋值，空间 $O(n)$；构造数据会退化。**


### 原理


### 实现

维护「颜色相同的连续段」集合，区间赋值时把重叠段合并，随机数据下均摊很快（$O(n\log\log n)$ 级别）。

```c++
struct Node { int l, r; mutable ll v; bool operator<(const Node& o) const { return l < o.l; } };
set<Node> odt;
auto split(int pos) {                        // 把含 pos 的段切成 [l,pos-1] 与 [pos,r]
    auto it = odt.lower_bound({pos, 0, 0});
    if (it != odt.end() && it->l == pos) return it;
    --it;
    int l = it->l, r = it->r; ll v = it->v;
    odt.erase(it);
    odt.insert({l, pos - 1, v});
    return odt.insert({pos, r, v}).first;
}
void assign_range(int l, int r, ll v) {
    auto itr = split(r + 1), itl = split(l);
    odt.erase(itl, itr);                     // 整段删掉
    odt.insert({l, r, v});
}
// 只能在「有大量随机区间赋值」的数据上保证复杂度；构造数据会退化
```

## 块状链表


### 一句话

维护一个序列（$n,m\le 10^5$），要不断在某个位置插入一个数、删除某个数、并把一段区间拿出来做查询，插入与查询交替出现。**时间单次操作 $O(\sqrt n)$，空间 $O(n)$。**


### 原理


### 实现

（把序列切成 $\sqrt n$ 块，每块存一个连续数组：定位 $O(\sqrt n)$、块内插入 $O(\sqrt n)$，块太大就裂开、块空了就删掉。）

```c++
// 块状链表：支持 O(sqrt n) 的「按下标插入 / 删除 / 查询」
const int B = 700;
vector<vector<int>> blk;

void build(int n, int* a) {
    blk.clear();
    for (int i = 0; i < n; i += B)
        blk.push_back(vector<int>(a + i, a + min(n, i + B)));
}
// 定位第 pos 个元素（从 0 开始），返回 (块号, 块内下标)
pair<int, int> locate(int pos) {
    for (int i = 0; i < (int)blk.size(); i++) {
        if (pos < (int)blk[i].size()) return {i, pos};
        pos -= blk[i].size();
    }
    return {(int)blk.size() - 1, (int)blk.back().size()};
}
void insert_at(int pos, int v) {
    auto pr = locate(pos);
    blk[pr.first].insert(blk[pr.first].begin() + pr.second, v);
    if ((int)blk[pr.first].size() > 2 * B) {                 // 太大了，裂成两块
        vector<int> nw(blk[pr.first].begin() + B, blk[pr.first].end());
        blk[pr.first].resize(B);
        blk.insert(blk.begin() + pr.first + 1, nw);
    }
}
void erase_at(int pos) {
    auto pr = locate(pos);
    blk[pr.first].erase(blk[pr.first].begin() + pr.second);
    if (blk[pr.first].empty()) blk.erase(blk.begin() + pr.first);   // 空了就删块
}
int query(int pos) { auto pr = locate(pos); return blk[pr.first][pr.second]; }

void traverse_all() {                        // 整体扫一遍仍是 O(n)
    for (auto& b : blk) for (int v : b) use(v);
}
// 块数始终是 O(sqrt n)（每块大小在 [B, 2B] 之间），所以 locate 与块内操作都是 O(sqrt n)。
// 若还要「区间加」，给每块再记一个 lazy 标记即可 —— 这就是块状数组 / 数列分块。
```

### 应用


## 莫队


### 一句话

给 $q$ 个区间询问（$n,q\le 10^5$），要你离线回答每个 $[l,r]$ 的答案，而当前区间的左右端点每移动一格、就把一个元素加进统计或移出统计，加与删交替出现。**时间总共 $O(n\sqrt q)$（普通莫队）、$O(n^{5/3})$（带修）、$O(n\sqrt q)$（回滚），空间 $O(n)$。**


### 原理


### 实现

```c++
int n, q, bs;
int a[N], cnt[N], curans, ans[N];
struct Qry { int l, r, id; };
bool operator<(const Qry& x, const Qry& y) {
    int bx = x.l / bs, by = y.l / bs;
    if (bx != by) return bx < by;
    return (bx & 1) ? x.r > y.r : x.r < y.r;  // 奇偶化排序，常数更小
}
void add_pos(int i) { if (++cnt[a[i]] == 1) curans++; }
void del_pos(int i) { if (--cnt[a[i]] == 0) curans--; }
void mo(vector<Qry>& qs) {
    bs = max(1, (int)(n / sqrt((double)qs.size())));
    sort(qs.begin(), qs.end());
    int l = 1, r = 0;
    for (auto& qq : qs) {
        while (l > qq.l) add_pos(--l);
        while (r < qq.r) add_pos(++r);
        while (l < qq.l) del_pos(l++);
        while (r > qq.r) del_pos(r--);
        ans[qq.id] = curans;
    }
}
```

#### 回滚莫队
只支持「加」不支持「删」时（例如维护最大值），把询问按左端点分块，右端点单调右移、左端点小范围回滚。

```c++
// 对每个左端点所在的块 b：
//   1. 把「右指针」从块尾 b*bs 一路右移（只加不删），得到 [b*bs+1, r] 的答案
//   2. 对块内每个询问 [l, r]（l 与 b*bs 同块）：
//        先记下当前答案；再把 [l, b*bs] 这一段暴力加进去
//        得到答案后，把这几个位置逐个撤销（回滚）
// 复杂度 O(n sqrt n)，全程只有「加」和「撤销」
void rollback_mo(vector<Qry>& qs, int n, int bs) {
    sort(qs.begin(), qs.end(), [](const Qry& a, const Qry& b) { return a.r < b.r; });
    for (int b = 1; b * bs <= n; b++) {
        int L = b * bs + 1, R = L - 1;        // 右区间的范围 [L, R]
        reset();
        for (auto& q : qs) {
            if (q.l > b * bs || q.l < (b - 1) * bs + 1) continue;   // 不在这个块
            while (R < q.r) add_pos(++R);     // 右指针只右移
            int save = curans, savepos = R;
            for (int i = min(q.r, b * bs); i >= q.l; i--) add_pos(i);   // 左边暴力加
            ans[q.id] = curans;
            while (R > savepos) del_pos(R--); // 撤销右侧（实际只需撤销左侧）
            for (int i = q.l; i <= min(q.r, b * bs); i++) del_pos(i);
            curans = save;
        }
    }
}
```


#### 树上莫队
用欧拉序把树上的路径问题转成序列区间问题：每条路径对应欧拉序上的两段（或一段），出现两次的点表示「不在路径上」。

```c++
    for (int v : g[u]) if (v != p) dfs_order(v, u);
```

#### 二次离线莫队

（把「指针移动时产生的贡献」拆成「前缀形式」，再用一次离线扫描回答所有前缀询问；代价是允许「$O(1)$ 修改、$O(1)$ 查前缀」，适用范围比普通莫队大得多。）

```c++
// 例：区间逆序对计数。记 f(x, i) = 「a[x] 与 a[1..i] 之间的逆序对数」。
// 右指针从 r 移到 r+1 时，新增贡献 = f(r+1, r) - f(r+1, l-1)：
//   前半 f(r+1, r) 是定值，可以在扫描时直接算；
//   后半是关于 l-1 的前缀询问，攒起来离线回答。
struct Ask { int pos, l, r, id, sgn; };       // 询问「sum_{i in [l,r]} f(i, pos)」

vector<Ask> bucket[N];                        // 按 pos 分桶
ll ans[N], pref[N];                           // pref：贡献的前缀和

void second_offline(int n, int m) {
    // 1) 对每个询问 [L, R]，把 l、r 两个指针的每次移动都拆成
    //    一条定值 + 一条形如 (pos, [l, r], sgn) 的前缀询问，塞进 bucket[pos]
    // 2) 从小到大扫一遍 i，维护「当前 i 之前的信息」；
    //    扫到 pos 时把 bucket[pos] 里的询问一次性回答（此时信息已经就位）
    // 3) 把回答累加回 ans
    for (int i = 1; i <= n; i++) {
        // add(i)：把 a[i] 插入当前数据结构（O(1) 修改）
        for (auto& q : bucket[i]) {
            // 查询 [q.l, q.r] 的贡献和（O(1) 查前缀）
            (void)q;
        }
    }
}
// 复杂度 O(n sqrt m + n * 修改代价 + m * 查询代价)，通常写作 O(n sqrt m)。
// 适用条件只有一条：贡献能写成「关于某个前缀的询问」，而不必要求 O(1) 增删。
```

## CDQ 分治

### 一句话

给你 $n$ 个三元组 $(a_i,b_i,c_i)$（$n\le 10^5$），要求出每个 $i$ 满足 $a_j\le a_i$、$b_j\le b_i$、$c_j\le c_i$ 的 $j$ 的个数（三维偏序）；也用来处理「先单点修改、再区间询问」这类带时间维的离线问题。**时间 $O(n\log^2 n)$，空间 $O(n)$。**

### 原理

### 实现

**CDQ 做「带时间维」的离线问题**（并入自 13 章）
把每个修改和询问都编成一个事件 $(t,p,v,ty)$：修改是 $ty=0$、值为 $v$；询问是 $ty=1$、$v$ 是正负号。分治按时间切，左半的修改去更新右半的询问，位置维用树状数组。

```c++
struct Event { int t, p, ty, id; ll v; };
Event ev[3 * N], buf[3 * N];
ll res[N];

void cdq_events(int l, int r) {
    if (l == r) return;
    int mid = (l + r) >> 1;
    cdq_events(l, mid), cdq_events(mid + 1, r);
    int i = l, j = mid + 1, k = l;
    while (j <= r) {                          // 左半的修改 + 右半的询问，按位置归并
        while (i <= mid && ev[i].p <= ev[j].p) {
            if (ev[i].ty == 0) add(ev[i].p, ev[i].v);
            buf[k++] = ev[i++];
        }
        if (ev[j].ty == 1) res[ev[j].id] += qry(ev[j].p) * ev[j].v;
        buf[k++] = ev[j++];
    }
    for (int t = l; t < i; t++)               // 撤销左半的修改
        if (ev[t].ty == 0) add(ev[t].p, -ev[t].v);
    while (i <= mid) buf[k++] = ev[i++];
    for (int t = l; t <= r; t++) ev[t] = buf[t];
}
// 用法：把「某时刻在 p 处 +v」写成 {t, p, 0, 0, v}，
//       把「时刻 t 询问前缀 [1, p] 的和」拆成两个询问 {t, p, 1, id, +1} 与 {t, l-1, 1, id, -1}
```

## 整体二分

> 【说明】下面「实现」一节保留了两段：第一段是最小骨架，第二段是可直接套用的完整写法。


### 一句话

给 $n,q\le 10^5$ 次操作 / 询问，要你离线回答每个询问的「第 $k$ 小」或带时间维的偏序计数，做法是把所有询问和答案值域一起二分、每次把修改按值域分到左右两边再判定，分与判交替出现。**时间 $O((n+q)\log n\log V)$，空间 $O(n+q)$。**


### 原理


### 实现

把「二分答案」整体化：所有询问一起二分，用树状数组统一结算贡献，$O((n+q)\log n\log V)$。

```c++
int n, q, a[N], ans[N];
struct Op { int l, r, k, id, type; };        // type 0：插入点，1：询问
vector<Op> ops;

void solve(int ql, int qr, int vl, int vr) {
    if (ql > qr) return;
    if (vl == vr) {
        for (int i = ql; i <= qr; i++) if (ops[i].type) ans[ops[i].id] = vl;
        return;
    }
    int mid = (vl + vr) >> 1;
    vector<Op> L, R;
    for (int i = ql; i <= qr; i++) {
        if (ops[i].type == 0) {              // 插入：值 <= mid 的进左边并加到 BIT
            if (ops[i].l <= mid) bit_add(ops[i].id, ops[i].k), L.push_back(ops[i]);
            else R.push_back(ops[i]);
        } else {
            int c = bit_sum(ops[i].r) - bit_sum(ops[i].l - 1);
            if (c >= ops[i].k) L.push_back(ops[i]);
            else ops[i].k -= c, R.push_back(ops[i]);
        }
    }
    for (auto& o : L) if (o.type == 0) bit_add(o.id, -o.k);   // 撤销 BIT
    int m = ql + L.size();
    for (size_t i = 0; i < L.size(); i++) ops[ql + i] = L[i];
    for (size_t i = 0; i < R.size(); i++) ops[m + i] = R[i];
    solve(ql, m - 1, vl, mid);
    solve(m, qr, mid + 1, vr);
}
// 与「线段树分治」互补：整体二分适合「答案可二分 + 贡献可差分」
```

#### 整体二分
一批询问一起二分答案：每次用 BIT 把「值落在左半」的位置标出来，按区间内计数把询问分流。下面是「静态区间第 $k$ 小」，单次分治三层结构，总复杂度 $O((n+q)\log V\log n)$。

```c++
int a[N], ansq[N];
struct Q { int l, r, k, id; };
void overall_bisect(int l, int r, vector<Q> q, vector<int> pos) {
    if (l == r) {
        for (auto& x : q) ansq[x.id] = l;
        return;
    }
    int mid = (l + r) >> 1;
    vector<int> pl, pr;
    for (int i : pos)
        if (a[i] <= mid) add(i, 1), pl.push_back(i);   // 只把左半的值放进 BIT
        else pr.push_back(i);
    vector<Q> ql, qr;
    for (auto& x : q) {
        int c = qry(x.r) - qry(x.l - 1);      // 区间里有多少个 <= mid 的数
        if (c >= x.k) ql.push_back(x);
        else x.k -= c, qr.push_back(x);
    }
    for (int i : pl) add(i, -1);              // 撤销
    overall_bisect(l, mid, ql, pl);
    overall_bisect(mid + 1, r, qr, pr);
}
// 调用：overall_bisect(1, maxv, queries, {1..n})，此时 BIT 的下标是位置而不是值
```

## K-D 树


### 一句话

给 $n$ 个 $k$ 维点（$n\le 10^5$，$k$ 通常 $2$ 或 $3$），要你不断插入点，并询问离某个点最近 / 第 $k$ 近的点是谁、或一个矩形里有多少个点，插入与查询交替出现。**时间建树 $O(n\log n)$、单次查询平均 $O(\sqrt n)$，空间 $O(n)$。**


### 原理


### 实现

（按维度轮换地取中位数分割，树上每个结点记录「子树内每个维度的坐标范围」用于剪枝。）

```c++
struct KD {
    int d[2], id;
    int mn[2], mx[2], ls, rs;
} t[N];
int rt, tot;
int dim;

bool cmp(const KD& a, const KD& b) { return a.d[dim] < b.d[dim]; }
void pushup(int o) {
    for (int k = 0; k < 2; k++) t[o].mn[k] = t[o].mx[k] = t[o].d[k];
    for (int s = 0; s < 2; s++) {
        int c = s ? t[o].rs : t[o].ls;
        if (!c) continue;
        for (int k = 0; k < 2; k++) {
            t[o].mn[k] = min(t[o].mn[k], t[c].mn[k]);
            t[o].mx[k] = max(t[o].mx[k], t[c].mx[k]);
        }
    }
}
int build(int l, int r, int dep) {
    if (l > r) return 0;
    int mid = (l + r) >> 1;
    dim = dep & 1;
    nth_element(t + l, t + mid, t + r + 1, cmp);
    int o = mid;
    t[o].ls = build(l, mid - 1, dep + 1);
    t[o].rs = build(mid + 1, r, dep + 1);
    pushup(o);
    return o;
}
ll dist2(const KD& a, int x, int y) { ll dx = a.d[0] - x, dy = a.d[1] - y; return dx * dx + dy * dy; }
ll mindist2(int o, int x, int y) {           // 点 (x,y) 到结点 o 所辖矩形的最小可能距离
    ll r = 0;
    for (int k = 0; k < 2; k++) {
        int c = (k == 0 ? x : y);
        if (c < t[o].mn[k]) r += (ll)(t[o].mn[k] - c) * (t[o].mn[k] - c);
        if (c > t[o].mx[k]) r += (ll)(c - t[o].mx[k]) * (c - t[o].mx[k]);
    }
    return r;
}
```

```c++
    ll d = dist2(t[o], x, y);
    if ((int)heap.size() < k) heap.push(d);
    else if (d < heap.top()) heap.pop(), heap.push(d);
```

## Link-Cut Tree


### 一句话

维护一片森林（$n$ 个点，$m$ 次操作，$n,m\le 10^5$），要不断在 $u$ 与 $v$ 之间连一条边、把某条边断开、问 $u$ 到 $v$ 路径上的和 / 最值、把根换成 $x$、并询问 $u$ 与 $v$ 现在是否连通，连边与查询交替出现。**时间单次操作均摊 $O(\log n)$，空间 $O(n)$。**

### 实现

（用 Splay 维护「实链」，每个 Splay 的中序对应原树上一条从上到下的路径；核心操作是 access。）

```c++
int ch[N][2], lct_fa[N], rev[N];
ll val[N], lct_sum[N];                       // val 是点权，sum 是 splay 子树和

bool isroot(int x) { return ch[lct_fa[x]][0] != x && ch[lct_fa[x]][1] != x; }
void pushup(int x) { lct_sum[x] = lct_sum[ch[x][0]] + lct_sum[ch[x][1]] + val[x]; }
void pushrev(int x) { if (!x) return; swap(ch[x][0], ch[x][1]); rev[x] ^= 1; }
void pushdown(int x) { if (rev[x]) { pushrev(ch[x][0]); pushrev(ch[x][1]); rev[x] = 0; } }
void rotate(int x) {
    int y = lct_fa[x], z = lct_fa[y];
    int k = (ch[y][1] == x), w = ch[x][k ^ 1];
    if (!isroot(y)) ch[z][ch[z][1] == y] = x;
    lct_fa[x] = z;
    ch[x][k ^ 1] = y, lct_fa[y] = x;
    ch[y][k] = w;
    if (w) lct_fa[w] = y;
    pushup(y), pushup(x);
}
void splay(int x) {
    static int stk[N];
    int top = 0, y = x;
    stk[++top] = y;
    while (!isroot(y)) stk[++top] = y = lct_fa[y];
    while (top) pushdown(stk[top--]);
    while (!isroot(x)) {
        int y = lct_fa[x], z = lct_fa[y];
        if (!isroot(y)) rotate(((ch[y][0] == x) ^ (ch[z][0] == y)) ? x : y);
        rotate(x);
    }
}
void access(int x) {
    for (int y = 0; x; y = x, x = lct_fa[x]) {
        splay(x);
        ch[x][1] = y;                        // 换实儿子
        pushup(x);
    }
}
void makeroot(int x) { access(x); splay(x); pushrev(x); }
int findroot(int x) {
    access(x); splay(x); pushdown(x);
    while (ch[x][0]) x = ch[x][0], pushdown(x);
    splay(x);
    return x;
}
void link(int x, int y) { makeroot(x); if (findroot(y) != x) lct_fa[x] = y; }
void cut(int x, int y) {
    makeroot(x);
    if (findroot(y) == x && lct_fa[y] == x && !ch[y][0]) ch[x][1] = 0, lct_fa[y] = 0, pushup(x);
}
void split(int x, int y) { makeroot(x); access(y); splay(y); }   // 路径 x..y 落在 y 的 splay 里
```


### 应用


#### 动态树上的路径查询与最小生成树
路径查询用 split(x,y) 之后直接读 y 的聚合值；动态 MST 用「每条边建一个点」的技巧。

```c++
ll path_sum(int x, int y) { split(x, y); return lct_sum[y]; }   // 路径和
// 动态 MST（只加边）：
//   把原图的点权设为 -inf，每条边也建一个结点、权值设为边权
//   加入边 (u,v,w) 时：
//     若 findroot(u) != findroot(v)：直接 link(u, e) 再 link(e, v)
//     否则查询路径 u..v 上的最大边权，若 > w 就把那条边对应结点断开、接上新边
ll mxv[N];
void pushup_mx(int x) { mxv[x] = max(val[x], max(mxv[ch[x][0]], mxv[ch[x][1]])); }
void pushup_full(int x) { pushup(x); pushup_mx(x); }   // 和与最值一起维护
int maxnode(int x, int y) {                  // 路径 x..y 上权值最大的结点
    split(x, y);
    int o = y, best = y;
    while (o) {
        if (val[o] > val[best]) best = o;
        pushdown(o);
        if (ch[o][1] && mxv[ch[o][1]] > val[best]) o = ch[o][1];
        else if (ch[o][0] && mxv[ch[o][0]] > val[best]) o = ch[o][0];
        else break;
    }
    return best;
}
```


#### 维护连通性（带断边）
用 findroot 判断两点是否在同一棵树里；断边用 cut（前提是这条边确实存在）。

```c++
bool connected(int x, int y) { return findroot(x) == findroot(y); }
// 注意：cut 之前最好先 makeroot(x) 再检查 lct_fa[y] == x 且 y 无左儿子，
// 否则会把本来不相邻的两点「切断」
```


#### LCT 维护子树信息
普通的 LCT 只维护「实链」上的信息；要维护子树信息需要额外记录「虚儿子的聚合值」，并在 access 时同步更新。

```c++
int ssz[N], allsz[N];                        // ssz：虚儿子的大小和；allsz：整棵子树大小
void pushup(int x) {
    lct_sum[x] = lct_sum[ch[x][0]] + lct_sum[ch[x][1]] + val[x] + ssz[x];
    allsz[x] = allsz[ch[x][0]] + allsz[ch[x][1]] + 1 + ssz[x];
}
void access(int x) {
    for (int y = 0; x; y = x, x = lct_fa[x]) {
        splay(x);
        // 旧右儿子从实转虚：加进 ssz
        if (ch[x][1]) ssz[x] += allsz[ch[x][1]];
        // 新右儿子从虚转实：从 ssz 减掉
        if (y) ssz[x] -= allsz[y];
        ch[x][1] = y;
        pushup(x);
    }
}
// 例：动态维护「每棵树的大小」，link 时 allsz 会自动合并
```
